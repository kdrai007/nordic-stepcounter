import com.nordic.stepcounter.ui.theme.LocalStepPalette
import com.nordic.stepcounter.ui.theme.NordicStepTheme

class MainActivity : ComponentActivity() {

    private val viewModel: StepViewModel by viewModels {
        StepViewModel.provideFactory(applicationContext)
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        setContent {
            val state by viewModel.uiState.collectAsState()
            var currentTab by rememberSaveable { mutableIntStateOf(0) }

            val permissionLauncher = rememberLauncherForActivityResult(
                contract = ActivityResultContracts.RequestMultiplePermissions()
            ) {
                startStepServiceIfPermitted()
            }

            LaunchedEffect(Unit) {
                if (!StepCounterService.hasActivityRecognitionPermission(this@MainActivity)) {
                    val permissionsToRequest = buildList {
                        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                            add(Manifest.permission.ACTIVITY_RECOGNITION)
                        }
                        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
                            add(Manifest.permission.POST_NOTIFICATIONS)
                        }
                    }
                    if (permissionsToRequest.isNotEmpty()) {
                        permissionLauncher.launch(permissionsToRequest.toTypedArray())
                    }
                } else {
                    startStepServiceIfPermitted()
                }
            }

            NordicStepTheme(appTheme = state.appTheme) {
                val palette = LocalStepPalette.current
                Scaffold(
                    containerColor = palette.background,
                    bottomBar = {
                        NavigationBar(containerColor = palette.surface) {
                            NavigationBarItem(
                                selected = currentTab == 0,
                                onClick = { currentTab = 0 },
                                icon = { Icon(painterResource(R.drawable.ic_tab_today), contentDescription = "Today") },
                                label = { Text("TODAY") }
                            )
                            NavigationBarItem(
                                selected = currentTab == 1,
                                onClick = { currentTab = 1 },
                                icon = { Icon(painterResource(R.drawable.ic_tab_history), contentDescription = "History") },
                                label = { Text("HISTORY") }
                            )
                            NavigationBarItem(
                                selected = currentTab == 2,
                                onClick = { currentTab = 2 },
                                icon = { Icon(painterResource(R.drawable.ic_tab_profile), contentDescription = "Profile") },
                                label = { Text("PROFILE") }
                            )
                        }
                    }
                ) { innerPadding ->
                    when (currentTab) {
                        0 -> TodayScreen(
                            state = state,
                            onTogglePause = viewModel::togglePause,
                            modifier = Modifier.padding(innerPadding)
                        )
                        1 -> HistoryScreen(
                            state = state,
                            modifier = Modifier.padding(innerPadding)
                        )
                        2 -> ProfileScreen(
                            state = state,
                            onSelectGoal = viewModel::updateGoal,
                            onSelectTheme = viewModel::selectTheme,
                            onUpdatePhysicalProfile = viewModel::updatePhysicalProfile,
                            modifier = Modifier.padding(innerPadding)
                        )
                    }
                }
            }
        }
    }

    private fun startStepServiceIfPermitted() {
        if (StepCounterService.hasActivityRecognitionPermission(this)) {
            val intent = Intent(this, StepCounterService::class.java)
            ContextCompat.startForegroundService(this, intent)
        }
    }
}

package com.nordic.stepcounter.data.local

import androidx.room.Dao
import androidx.room.Query
import androidx.room.Upsert
import kotlinx.coroutines.flow.Flow

@Dao
interface StepDao {
    @Query("SELECT * FROM daily_steps WHERE date = :date LIMIT 1")
    fun getDailyRecord(date: String): Flow<DailyStepEntity?>

    @Query("SELECT * FROM daily_steps WHERE date = :date LIMIT 1")
    suspend fun getDailyRecordDirect(date: String): DailyStepEntity?

    @Query("SELECT * FROM daily_steps ORDER BY date DESC LIMIT :limit")
    fun getRecentDailyRecords(limit: Int): Flow<List<DailyStepEntity>>

    @Query("SELECT * FROM hourly_steps WHERE date = :date ORDER BY hour ASC")
    fun getHourlyRecordsForDate(date: String): Flow<List<HourlyStepEntity>>

    @Query("SELECT * FROM hourly_steps WHERE date = :date AND hour = :hour LIMIT 1")
    suspend fun getHourlyRecordDirect(date: String, hour: Int): HourlyStepEntity?

    @Upsert
    suspend fun upsertDaily(entity: DailyStepEntity)

    @Upsert
    suspend fun upsertHourly(entity: HourlyStepEntity)

    @Query("UPDATE daily_steps SET goalSteps = :goal WHERE date = :date")
    suspend fun updateDailyGoal(date: String, goal: Int): Int

    @Query("SELECT SUM(steps) FROM daily_steps")
    suspend fun getTotalLifetimeSteps(): Long?
}

package com.nordic.stepcounter.data.local

import android.content.Context

import androidx.room.Room
import androidx.room.RoomDatabase

@Database(entities = [DailyStepEntity::class, HourlyStepEntity::class], version = 1, exportSchema = false)
abstract class StepDatabase : RoomDatabase() {
    abstract fun stepDao(): StepDao

    companion object {
        @Volatile
        private var INSTANCE: StepDatabase? = null

        fun getInstance(context: Context): StepDatabase {
            return INSTANCE ?: synchronized(this) {
                INSTANCE ?: Room.databaseBuilder(
                    context.applicationContext,
                    StepDatabase::class.java,
                    "nordic_steps.db"
                ).build().also { INSTANCE = it }
            }
        }
    }
}

data class UserProfile(
    val heightCm: Float = 175.0f,
    val weightKg: Float = 70.0f,
    val isMale: Boolean = true,
    val customStrideLengthMeters: Float = 0.0f,
    val dailyGoal: Int = 6000,
    val isPaused: Boolean = false,
    val lastSensorStepValue: Long = -1L,
    val lastRecordedDate: String = "",
    val appTheme: AppTheme = AppTheme.NORDIC_DARK
) {
    val effectiveStrideLengthMeters: Float
        get() = if (customStrideLengthMeters > 0f) {
            customStrideLengthMeters
        } else {
            StepCalculationEngine.calculateDefaultStrideLengthMeters(heightCm, isMale)
        }

    val effectiveStrideMeters: Float
        get() = effectiveStrideLengthMeters
}

private val Context.dataStore: DataStore<Preferences> by preferencesDataStore(name = "nordic_user_preferences")

class ProfilePreferences(
    private val dataStore: DataStore<Preferences>
) {
    constructor(
        context: Context,
        dataStore: DataStore<Preferences> = context.applicationContext.dataStore
    ) : this(dataStore)

    private object PreferencesKeys {
        val KEY_HEIGHT_CM = floatPreferencesKey("key_height_cm")
        val KEY_WEIGHT_KG = floatPreferencesKey("key_weight_kg")
        val KEY_IS_MALE = booleanPreferencesKey("key_is_male")
        val KEY_CUSTOM_STRIDE_METERS = floatPreferencesKey("key_custom_stride_meters")
        val KEY_DAILY_GOAL = intPreferencesKey("key_daily_goal")
        val KEY_IS_PAUSED = booleanPreferencesKey("key_is_paused")
        val KEY_LAST_SENSOR_STEP_VALUE = longPreferencesKey("key_last_sensor_step_value")
        val KEY_LAST_RECORDED_DATE = stringPreferencesKey("key_last_recorded_date")
        val KEY_APP_THEME = stringPreferencesKey("key_app_theme")
    }

    val userProfileFlow: Flow<UserProfile> = dataStore.data
        .catch { exception ->
            if (exception is IOException) {
                emit(emptyPreferences())
            } else {
                throw exception
            }
        }
        .map { preferences ->
            val themeName = preferences[PreferencesKeys.KEY_APP_THEME]
            val appTheme = themeName?.let {
                runCatching { AppTheme.valueOf(it) }.getOrNull()
            } ?: AppTheme.NORDIC_DARK

            UserProfile(
                heightCm = preferences[PreferencesKeys.KEY_HEIGHT_CM] ?: 175.0f,
                weightKg = preferences[PreferencesKeys.KEY_WEIGHT_KG] ?: 70.0f,
                isMale = preferences[PreferencesKeys.KEY_IS_MALE] ?: true,
                customStrideLengthMeters = preferences[PreferencesKeys.KEY_CUSTOM_STRIDE_METERS] ?: 0.0f,
                dailyGoal = preferences[PreferencesKeys.KEY_DAILY_GOAL] ?: 6000,
                isPaused = preferences[PreferencesKeys.KEY_IS_PAUSED] ?: false,
                lastSensorStepValue = preferences[PreferencesKeys.KEY_LAST_SENSOR_STEP_VALUE] ?: -1L,
                lastRecordedDate = preferences[PreferencesKeys.KEY_LAST_RECORDED_DATE] ?: "",
                appTheme = appTheme
            )
        }

    val isPausedFlow: Flow<Boolean> = userProfileFlow.map { it.isPaused }.distinctUntilChanged()
    val dailyGoalFlow: Flow<Int> = userProfileFlow.map { it.dailyGoal }.distinctUntilChanged()
    val appThemeFlow: Flow<AppTheme> = userProfileFlow.map { it.appTheme }.distinctUntilChanged()

    suspend fun setPaused(isPaused: Boolean) {
        dataStore.edit { preferences ->
            preferences[PreferencesKeys.KEY_IS_PAUSED] = isPaused
        }
    }

    suspend fun updateDailyGoal(goal: Int) {
        dataStore.edit { preferences ->
            preferences[PreferencesKeys.KEY_DAILY_GOAL] = goal
        }
    }

    suspend fun updatePhysicalProfile(
        heightCm: Float,
        weightKg: Float,
        isMale: Boolean,
        customStrideLengthMeters: Float = 0.0f
    ) {
        dataStore.edit { preferences ->
            preferences[PreferencesKeys.KEY_HEIGHT_CM] = heightCm
            preferences[PreferencesKeys.KEY_WEIGHT_KG] = weightKg
            preferences[PreferencesKeys.KEY_IS_MALE] = isMale
            preferences[PreferencesKeys.KEY_CUSTOM_STRIDE_METERS] = customStrideLengthMeters
        }
    }

    suspend fun updateProfile(
        heightCm: Float,
        weightKg: Float,
        isMale: Boolean,
        customStrideMeters: Float = 0.0f
    ) {
        updatePhysicalProfile(
            heightCm = heightCm,
            weightKg = weightKg,
            isMale = isMale,
            customStrideLengthMeters = customStrideMeters
        )
    }

    suspend fun updateLastSensorReading(steps: Long, date: String) {
        dataStore.edit { preferences ->
            preferences[PreferencesKeys.KEY_LAST_SENSOR_STEP_VALUE] = steps
            preferences[PreferencesKeys.KEY_LAST_RECORDED_DATE] = date
        }
    }

    suspend fun updateLastSensorStepValue(stepValue: Long) {
        dataStore.edit { preferences ->
            preferences[PreferencesKeys.KEY_LAST_SENSOR_STEP_VALUE] = stepValue
        }
    }

    suspend fun updateAppTheme(theme: AppTheme) {
        dataStore.edit { preferences ->
            preferences[PreferencesKeys.KEY_APP_THEME] = theme.name
        }
    }

    suspend fun clear() {
        dataStore.edit { preferences ->
            preferences.clear()
        }
    }

    companion object {
        @Volatile
        private var INSTANCE: ProfilePreferences? = null

        fun getInstance(context: Context): ProfilePreferences {
            return INSTANCE ?: synchronized(this) {
                INSTANCE ?: ProfilePreferences(context.applicationContext.dataStore).also { INSTANCE = it }
            }
        }
    }
}

class StepRepository(

    val userProfileFlow: Flow<UserProfile> = preferences.userProfileFlow
    val isPausedFlow: Flow<Boolean> = preferences.isPausedFlow
    val dailyGoalFlow: Flow<Int> = preferences.dailyGoalFlow
    val appThemeFlow: Flow<AppTheme> = preferences.appThemeFlow

        get() = getDailyRecordFlow(LocalDate.now().toString())

    fun getLast7DaysRecordsFlow(): Flow<List<DailyStepEntity>> =
        getRecentDailyRecordsFlow(7)

        getRecentDailyRecordsFlow(30)

    fun getHourlyRecordsForTodayFlow(): Flow<List<HourlyStepEntity>> =
        getHourlyRecordsFlow(LocalDate.now().toString())

    fun getDailyRecordFlow(date: String): Flow<DailyStepEntity?> =
        stepDao.getDailyRecord(date)

    suspend fun getDailyRecordDirect(date: String): DailyStepEntity? =
        stepDao.getDailyRecordDirect(date)

    fun getRecentDailyRecordsFlow(limit: Int): Flow<List<DailyStepEntity>> =
        stepDao.getRecentDailyRecords(limit)

    fun getHourlyRecordsFlow(date: String): Flow<List<HourlyStepEntity>> =
        stepDao.getHourlyRecordsForDate(date)

    suspend fun getTotalLifetimeSteps(): Long =
        stepDao.getTotalLifetimeSteps() ?: 0L

    /**
     * Initializes a record for [date] if one does not already exist.
     * Ensures midnight rollover or new day start has a valid 0-step baseline.
     */
    suspend fun initializeTodayRecord(
        date: String = LocalDate.now().toString()
    ): DailyStepEntity = sensorMutex.withLock {
        val existing = stepDao.getDailyRecordDirect(date)
        if (existing != null) return@withLock existing

        val profile = preferences.userProfileFlow.first()
        val initial = DailyStepEntity(
            date = date,
            steps = 0,
            distanceMeters = 0f,
            caloriesKcal = 0f,
            activeDurationSeconds = 0L,
            goalSteps = profile.dailyGoal
        )
        stepDao.upsertDaily(initial)
        return@withLock initial
    }

    /**
     * Ingests a raw hardware Sensor.TYPE_STEP_COUNTER reading.
     * Handles delta calculation, midnight rollover, pause state, and updates
     * Room daily & hourly step records alongside DataStore sensor baseline.
     */
    suspend fun processHardwareSensorReading(
        rawSensorSteps: Long,
        date: String = LocalDate.now().toString(),
        hour: Int = LocalTime.now().hour
    ): DailyStepEntity = sensorMutex.withLock {
        val profile = preferences.userProfileFlow.first()

        val deltaResult = StepDeltaProcessor.computeDelta(
            currentSensorSteps = rawSensorSteps,
            lastSensorSteps = profile.lastSensorStepValue,
            isPaused = profile.isPaused
        )

        // Advance baseline and record latest date
        preferences.updateLastSensorReading(
            steps = deltaResult.newBaselineSensorSteps,
            date = date
        )

        val existingDaily = stepDao.getDailyRecordDirect(date)
        val baseSteps = existingDaily?.steps ?: 0
        val newTotalSteps = baseSteps + deltaResult.stepsToAdd

        val strideMeters = profile.effectiveStrideLengthMeters
        val distanceMeters = StepCalculationEngine.calculateDistanceKm(newTotalSteps, strideMeters) * 1000f
        val caloriesKcal = StepCalculationEngine.calculateCaloriesKcal(newTotalSteps, profile.weightKg)
        val activeDurationSeconds = StepCalculationEngine.calculateActiveMinutes(newTotalSteps) * 60L

        val updatedDaily = DailyStepEntity(
            date = date,
            steps = newTotalSteps,
            distanceMeters = distanceMeters,
            caloriesKcal = caloriesKcal,
            activeDurationSeconds = activeDurationSeconds,
            goalSteps = profile.dailyGoal
        )
        stepDao.upsertDaily(updatedDaily)

        if (deltaResult.stepsToAdd > 0) {
            val existingHourly = stepDao.getHourlyRecordDirect(date, hour)
            val newHourlySteps = (existingHourly?.steps ?: 0) + deltaResult.stepsToAdd
            stepDao.upsertHourly(
                HourlyStepEntity(
                    date = date,
                    hour = hour,
                    steps = newHourlySteps
                )
            )
        }

        return@withLock updatedDaily
    }

    suspend fun setPaused(isPaused: Boolean) {
        preferences.setPaused(isPaused)
    }

    suspend fun updateDailyGoal(goal: Int) = sensorMutex.withLock {
        preferences.updateDailyGoal(goal)
        val today = LocalDate.now().toString()
        val affected = stepDao.updateDailyGoal(today, goal)
        if (affected == 0) {
            val existing = stepDao.getDailyRecordDirect(today)
            if (existing == null) {
                stepDao.upsertDaily(
                    DailyStepEntity(
                        date = today,
                        steps = 0,
                        distanceMeters = 0f,
                        caloriesKcal = 0f,
                        activeDurationSeconds = 0L,
                        goalSteps = goal
                    )
                )
            }
        }
    }

    suspend fun updatePhysicalProfile(
        heightCm: Float,
        weightKg: Float,
        isMale: Boolean,
        customStrideLengthMeters: Float = 0.0f
    ) {
        preferences.updatePhysicalProfile(heightCm, weightKg, isMale, customStrideLengthMeters)
    }

    suspend fun updateAppTheme(theme: AppTheme) {
        preferences.updateAppTheme(theme)
    }

    companion object {
        @Volatile
        private var INSTANCE: StepRepository? = null

        fun getInstance(context: Context): StepRepository {
            return INSTANCE ?: synchronized(this) {
                INSTANCE ?: StepRepository(
                    StepDatabase.getInstance(context).stepDao(),
                    ProfilePreferences.getInstance(context)
                ).also { INSTANCE = it }
            }
        }

        fun getInstance(stepDao: StepDao, preferences: ProfilePreferences): StepRepository {
            return INSTANCE ?: synchronized(this) {
                INSTANCE ?: StepRepository(stepDao, preferences).also { INSTANCE = it }
            }
        }

        fun resetInstance() {
            synchronized(this) {
                INSTANCE = null
            }
        }
    }
}

package com.nordic.stepcounter.domain

/**
 * Pure domain calculation engine for physical fitness metrics:
 * Stride length, distance, active calories, and active duration.
 */
object StepCalculationEngine {
    private const val MALE_STRIDE_RATIO = 0.415f
    private const val FEMALE_STRIDE_RATIO = 0.413f
    private const val CALORIE_MET_FACTOR = 0.00075f
    private const val ESTIMATED_STEPS_PER_MINUTE = 100

    fun calculateDefaultStrideLengthMeters(heightCm: Float, isMale: Boolean): Float {
        if (!heightCm.isFinite() || heightCm <= 0f) return 0f
        val ratio = if (isMale) MALE_STRIDE_RATIO else FEMALE_STRIDE_RATIO
        return (heightCm * ratio) / 100f
    }

    fun calculateDistanceKm(steps: Int, strideMeters: Float): Float {
        if (steps <= 0 || !strideMeters.isFinite() || strideMeters <= 0f) return 0f
        return (steps * strideMeters) / 1000f
    }

    fun calculateCaloriesKcal(steps: Int, weightKg: Float): Float {
        if (steps <= 0 || !weightKg.isFinite() || weightKg <= 0f) return 0f
        return steps * weightKg * CALORIE_MET_FACTOR
    }

    fun calculateActiveMinutes(steps: Int): Int {
        if (steps <= 0) return 0
        return steps / ESTIMATED_STEPS_PER_MINUTE
    }
}

package com.nordic.stepcounter.domain

data class StepDeltaResult(
    val stepsToAdd: Int,
    val newBaselineSensorSteps: Long
)

/**
 * Resolves cumulative Sensor.TYPE_STEP_COUNTER values into valid step increments,
 * handling first initialization (-1L), device reboots, and pause states.
 */
object StepDeltaProcessor {

    fun computeDelta(
        currentSensorSteps: Long,
        lastSensorSteps: Long,
        isPaused: Boolean
    ): StepDeltaResult {
        if (currentSensorSteps < 0L) {
            return StepDeltaResult(
                stepsToAdd = 0,
                newBaselineSensorSteps = lastSensorSteps
            )
        }

        // First sensor reading ever recorded
        if (lastSensorSteps < 0L) {
            return StepDeltaResult(
                stepsToAdd = 0,
                newBaselineSensorSteps = currentSensorSteps
            )
        }

        // While paused, advance baseline so steps taken during pause are discarded
        if (isPaused) {
            return StepDeltaResult(
                stepsToAdd = 0,
                newBaselineSensorSteps = currentSensorSteps
            )
        }

        // Device rebooted since last reading (hardware counter reset to 0)
        val rawDelta = if (currentSensorSteps < lastSensorSteps) {
            currentSensorSteps
        } else {
            currentSensorSteps - lastSensorSteps
        }

        return StepDeltaResult(
            stepsToAdd = rawDelta.coerceIn(0L, Int.MAX_VALUE.toLong()).toInt(),
            newBaselineSensorSteps = currentSensorSteps
        )
    }
}

/**
 * BroadcastReceiver responsible for restarting the [StepCounterService] upon device reboot
 * or app package update. Verifies ACTIVITY_RECOGNITION runtime permission before launching
 * the Foreground Service to comply with Android 14/15 Foreground Service launch restrictions.
 */
class BootReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent?) {
        val action = intent?.action ?: return
        if (action == Intent.ACTION_BOOT_COMPLETED || action == Intent.ACTION_MY_PACKAGE_REPLACED) {
            if (StepCounterService.hasActivityRecognitionPermission(context)) {
                val serviceIntent = Intent(context, StepCounterService::class.java).apply {
                    this.action = StepCounterService.ACTION_START
                }
                ContextCompat.startForegroundService(context, serviceIntent)
            }
        }
    }
}

class StepCounterService : Service() {

    companion object {
        const val CHANNEL_ID = "step_counter_channel"
        const val NOTIFICATION_ID = 1001

        const val ACTION_START = "com.nordic.stepcounter.ACTION_START"
        const val ACTION_PAUSE = "com.nordic.stepcounter.ACTION_PAUSE"
        const val ACTION_RESUME = "com.nordic.stepcounter.ACTION_RESUME"
        const val ACTION_STOP = "com.nordic.stepcounter.ACTION_STOP"
        const val ACTION_TOGGLE_PAUSE = "com.nordic.stepcounter.ACTION_TOGGLE_PAUSE"

        fun hasActivityRecognitionPermission(context: Context): Boolean {
            return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                ContextCompat.checkSelfPermission(
                    context,
                    Manifest.permission.ACTIVITY_RECOGNITION
                ) == PackageManager.PERMISSION_GRANTED
            } else {
                true
            }
        }
    }

    internal var serviceScope: CoroutineScope = CoroutineScope(SupervisorJob() + Dispatchers.IO)
    internal lateinit var repository: StepRepository
    internal lateinit var sensorManager: StepSensorManager
    private var isForegroundStarted = false

    override fun onCreate() {
        super.onCreate()
        if (!hasActivityRecognitionPermission(this)) {
            stopSelf()
            return
        }

        repository = StepRepository.getInstance(this)

        createNotificationChannel()

        sensorManager = StepSensorManager(this) { rawSteps ->
            serviceScope.launch {
                val updatedDaily = repository.processHardwareSensorReading(rawSteps)
                val profile = repository.userProfileFlow.first()
                updateNotification(updatedDaily.steps, updatedDaily.goalSteps, profile.isPaused)
            }
        }

        serviceScope.launch {
            repository.userProfileFlow
                .map { it.isPaused to it.dailyGoal }
                .distinctUntilChanged()
                .collect {
                    if (isForegroundStarted) {
                        updateNotificationForCurrentState()
                    }
                }
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        if (!hasActivityRecognitionPermission(this)) {
            stopSelf()
            return START_NOT_STICKY
        }

        if (intent?.action != ACTION_STOP && !isForegroundStarted) {
            startTracking()
        }

        when (intent?.action) {
            ACTION_STOP -> {
                handleStop()
                return START_NOT_STICKY
            }
            ACTION_PAUSE -> {
                serviceScope.launch {
                    if (::sensorManager.isInitialized) {
                        sensorManager.flush()
                    }
                    repository.setPaused(true)
                    updateNotificationForCurrentState()
                }
            }
            ACTION_RESUME -> {
                serviceScope.launch {
                    if (::sensorManager.isInitialized) {
                        sensorManager.flush()
                    }
                    repository.setPaused(false)
                    updateNotificationForCurrentState()
                }
            }
            ACTION_TOGGLE_PAUSE -> {
                serviceScope.launch {
                    if (::sensorManager.isInitialized) {
                        sensorManager.flush()
                    }
                    val profile = repository.userProfileFlow.first()
                    val newPaused = !profile.isPaused
                    repository.setPaused(newPaused)
                    updateNotificationForCurrentState()
                }
            }
            ACTION_START, null -> {
                startTracking()
            }
            else -> {
                startTracking()
            }
        }

        return START_STICKY
    }

    private fun startTracking() {
        if (!sensorManager.isSensorAvailable) {
            handleStop()
            return
        }

        if (!isForegroundStarted) {
            val initialNotification = buildNotification(steps = 0, goal = 6000, isPaused = false)
            promoteToForeground(initialNotification)
            isForegroundStarted = true
        }

        sensorManager.startListening()

        serviceScope.launch {
            repository.initializeTodayRecord()
            updateNotificationForCurrentState()
        }
    }

    private fun handleStop() {
        if (::sensorManager.isInitialized) {
            sensorManager.stopListening()
        }
        ServiceCompat.stopForeground(this, ServiceCompat.STOP_FOREGROUND_REMOVE)
        isForegroundStarted = false
        stopSelf()
    }

    private fun promoteToForeground(notification: Notification) {
        val serviceType = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.UPSIDE_DOWN_CAKE) {
            ServiceInfo.FOREGROUND_SERVICE_TYPE_HEALTH
        } else {
            0
        }
        ServiceCompat.startForeground(
            this,
            NOTIFICATION_ID,
            notification,
            serviceType
        )
    }

    internal suspend fun updateNotificationForCurrentState() {
        val today = LocalDate.now().toString()
        val todayRecord = repository.getDailyRecordDirect(today)
        val profile = repository.userProfileFlow.first()
        val steps = todayRecord?.steps ?: 0
        val goal = profile.dailyGoal
        val isPaused = profile.isPaused
        updateNotification(steps, goal, isPaused)
    }

    internal fun updateNotification(steps: Int, goal: Int, isPaused: Boolean) {
        val notificationManager = getSystemService(NotificationManager::class.java)
        notificationManager?.notify(NOTIFICATION_ID, buildNotification(steps, goal, isPaused))
    }

    internal fun buildNotification(steps: Int, goal: Int, isPaused: Boolean): Notification {
        val openAppIntent = PendingIntent.getActivity(
            this,
            0,
            Intent(this, MainActivity::class.java),
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val actionIntent = Intent(this, StepCounterService::class.java).apply {
            action = if (isPaused) ACTION_RESUME else ACTION_PAUSE
        }
        val actionPendingIntent = PendingIntent.getService(
            this,
            1,
            actionIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val percent = if (goal > 0) ((steps * 100) / goal).coerceAtMost(100) else 0
        val statusText = if (isPaused) "Paused • $percent% of goal" else "$percent% of $goal goal"
        val actionLabel = if (isPaused) "Resume" else "Pause"

        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setSmallIcon(android.R.drawable.ic_menu_directions)
            .setContentTitle("$steps STEPS")
            .setContentText(statusText)
            .setOngoing(true)
            .setSilent(true)
            .setContentIntent(openAppIntent)
            .addAction(0, actionLabel, actionPendingIntent)
            .build()
    }

    private fun createNotificationChannel() {
        val channel = NotificationChannel(
            CHANNEL_ID,
            "Step Tracking Service",
            NotificationManager.IMPORTANCE_LOW
        ).apply {
            description = "Shows live daily step count and goal progress"
            setShowBadge(false)
        }
        val notificationManager = getSystemService(NotificationManager::class.java)
        notificationManager?.createNotificationChannel(channel)
    }

    override fun onDestroy() {
        if (::sensorManager.isInitialized) {
            sensorManager.stopListening()
        }
        serviceScope.cancel()
        super.onDestroy()
    }

    override fun onBind(intent: Intent?): IBinder? = null
}

package com.nordic.stepcounter.service

import android.content.Context
import android.hardware.Sensor
import android.hardware.SensorEvent
import android.hardware.SensorEventListener
import android.hardware.SensorManager
import kotlinx.coroutines.channels.BufferOverflow
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow

/**
 * Low-power hardware step counter manager.
 * Exclusively uses [Sensor.TYPE_STEP_COUNTER] (sensor type 19) with 60-second hardware
 * FIFO batching to minimize application and CPU wakeups.
 */
class StepSensorManager(
    private val sensorManager: SensorManager?,
    private val onStepCountChanged: ((Long) -> Unit)? = null
) : SensorEventListener {

    constructor(
        context: Context,
        onStepCountChanged: ((Long) -> Unit)? = null
    ) : this(
        context.getSystemService(Context.SENSOR_SERVICE) as? SensorManager,
        onStepCountChanged
    )

    private val stepCounterSensor: Sensor? =
        sensorManager?.getDefaultSensor(Sensor.TYPE_STEP_COUNTER)

    private val _stepEvents = MutableSharedFlow<Long>(
        replay = 1,
        extraBufferCapacity = 64,
        onBufferOverflow = BufferOverflow.DROP_OLDEST
    )
    val stepEvents: SharedFlow<Long> = _stepEvents.asSharedFlow()

    val isSensorAvailable: Boolean
        get() = stepCounterSensor != null

    /**
     * Registers the hardware step counter listener with 60-second report latency.
     * Returns true if registered successfully, false otherwise.
     */
    fun startListening(): Boolean {
        val sensor = stepCounterSensor ?: return false
        return sensorManager?.registerListener(
            this,
            sensor,
            SensorManager.SENSOR_DELAY_NORMAL,
            BATCH_LATENCY_US
        ) ?: false
    }

    /**
     * Unregisters the sensor listener.
     */
    fun stopListening() {
        sensorManager?.unregisterListener(this)
    }

    /**
     * Flushes the hardware sensor FIFO buffer to immediately process batched steps.
     * Returns true if flush was initiated, false otherwise.
     */
    fun flush(): Boolean {
        return sensorManager?.flush(this) ?: false
    }

    override fun onSensorChanged(event: SensorEvent?) {
        if (event == null) return
        if (event.sensor?.type == Sensor.TYPE_STEP_COUNTER) {
            val totalStepsSinceBoot = event.values.getOrNull(0)?.toLong() ?: return
            onStepCountChanged?.invoke(totalStepsSinceBoot)
            _stepEvents.tryEmit(totalStepsSinceBoot)
        }
    }

    override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) {
        // Step counter accuracy does not affect cumulative counting
    }

    companion object {
        const val BATCH_LATENCY_US = 60_000_000 // 60 seconds hardware batching latency
    }
}

package com.nordic.stepcounter.ui.components

import androidx.compose.foundation.Canvas
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.foundation.layout.height
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.geometry.CornerRadius
import androidx.compose.ui.geometry.Offset
import androidx.compose.ui.geometry.Size
import androidx.compose.ui.unit.dp
import com.nordic.stepcounter.ui.theme.LocalStepPalette

@Composable
fun SwissBarChart(
    values: List<Int>,
    maxValue: Int,
    modifier: Modifier = Modifier
) {
    val palette = LocalStepPalette.current
    val safeMax = maxValue.coerceAtLeast(1)

    Canvas(
        modifier = modifier
            .fillMaxWidth()
            .height(180.dp)
    ) {
        if (values.isEmpty()) return@Canvas
        val barCount = values.size
        val spacing = if (barCount > 15) 3.dp.toPx() else 8.dp.toPx()
        val totalSpacing = spacing * (barCount - 1)
        val maxBarWidth = 36.dp.toPx()
        val calculatedBarWidth = ((size.width - totalSpacing) / barCount).coerceAtLeast(3f)
        val barWidth = calculatedBarWidth.coerceAtMost(maxBarWidth)
        val totalContentWidth = barCount * barWidth + totalSpacing
        val startOffset = ((size.width - totalContentWidth) / 2f).coerceAtLeast(0f)

        values.forEachIndexed { index, value ->
            val ratio = (value.toFloat() / safeMax.toFloat()).coerceIn(0.04f, 1f)
            val barHeight = size.height * ratio
            val x = startOffset + index * (barWidth + spacing)
            val y = size.height - barHeight

            drawRoundRect(
                color = if (value > 0) palette.accent else palette.trackBackground,
                topLeft = Offset(x, y),
                size = Size(barWidth, barHeight),
                cornerRadius = CornerRadius(4.dp.toPx(), 4.dp.toPx())
            )
        }
    }
}

package com.nordic.stepcounter.ui.theme

import android.app.Activity
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.darkColorScheme
import androidx.compose.material3.lightColorScheme
import androidx.compose.runtime.Composable
import androidx.compose.runtime.CompositionLocalProvider
import androidx.compose.runtime.SideEffect
import androidx.compose.ui.platform.LocalView
import androidx.core.view.WindowCompat
import com.nordic.stepcounter.data.preferences.AppTheme

@Composable
fun NordicStepTheme(
    appTheme: AppTheme = AppTheme.NORDIC_DARK,
    content: @Composable () -> Unit
) {
    val palette = appTheme.toStepColorPalette()

    val colorScheme = if (appTheme.isDark) {
        darkColorScheme(
            background = palette.background,
            surface = palette.surface,
            primary = palette.accent,
            onBackground = palette.primaryText,
            onSurface = palette.primaryText,
            outline = palette.border
        )
    } else {
        lightColorScheme(
            background = palette.background,
            surface = palette.surface,
            primary = palette.accent,
            onBackground = palette.primaryText,
            onSurface = palette.primaryText,
            outline = palette.border
        )
    }

    val view = LocalView.current
    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as? Activity)?.window ?: return@SideEffect
            val insetsController = WindowCompat.getInsetsController(window, view)
            insetsController.isAppearanceLightStatusBars = !appTheme.isDark
            insetsController.isAppearanceLightNavigationBars = !appTheme.isDark
        }
    }

    CompositionLocalProvider(LocalStepPalette provides palette) {
        MaterialTheme(
            colorScheme = colorScheme,
            typography = SwissTypography,
            content = content
        )
    }
}

        val args = params.map { type ->
            when (type) {
                Int::class.javaPrimitiveType -> 16
                Long::class.javaPrimitiveType -> 0L
                FloatArray::class.java -> FloatArray(16)
                Sensor::class.java -> mockk<Sensor>(relaxed = true)

        valuesField.isAccessible = true

        every { mockSensor.type } returns sensorType

# Nordic Step Counter - Architectural Design Specification

**Date:** 2026-11-10  
**Project:** Nordic Step Counter (Clean-room Android Pedometer)  
**Target Package:** `com.nordic.stepcounter`  
**Reference App Analyzed:** `steptracker.healthandfitness.walkingtracker.pedometer` (44.5 MB)

---

## 1. Problem & Context

Commercial pedometer applications (such as the analyzed `Step.apk`) provide useful daily tracking features but are heavily weighed down by:

- Over 31 MB of multi-dex code and dynamic DEX assets dedicated to AdMob, Meta Audience Network, and tracking SDKs.
- Continuous background telemetry services (`AppMeasurementService`, `SessionLifecycleService`, `CctBackendFactory`).
- Invasive permissions including background fine/coarse GPS location (`ACCESS_BACKGROUND_LOCATION`) and Google Advertising IDs (`AD_ID`, `ACCESS_ADSERVICES_AD_ID`).

The goal of this project is to build a modern, high-performance, battery-efficient, and privacy-respecting Android step tracker inspired by the core feature set of `Step.apk`, engineered with a **Minimalist Nordic / Swiss** typographic layout and a **Multi-Theme Palette Engine** (Nordic, Gruvbox, Catppuccin, and TokyoNight in both Dark and Light variants).

| :-------------------------- | :------------------------------------------------------------------- | :-------------------------------------------------------------- |
| **Download / Install Size** | 44.5 MB (4 DEX files + dynamic asset DEX)                            | ~3 - 5 MB (Single DEX, stripped dependencies)                   |
| **Ad Networks & SDKs**      | Google AdMob, Meta Audience Network, AppLovin                        | **None (Zero ads)**                                             |

## 4. Hardware Sensor Engine, State Math & Android 14/15 Compliance

1. **Low-Power Hardware Batching (`Sensor.TYPE_STEP_COUNTER`):**
   - Android's `Sensor.TYPE_STEP_COUNTER` (sensor ID 19) operates inside the device's low-power sensor hub DSP. It counts steps continuously without keeping the main Application Processor (CPU) awake.
   - `StepSensorManager` registers the listener with `maxReportLatencyUs = 60_000_000` (60 seconds) so the hardware buffers steps in batches while the screen is off. `Sensor.TYPE_STEP_DETECTOR` (18) is intentionally omitted to avoid unbatched per-step CPU wakeups.
2. **Cumulative Delta, Reboot, Pause & Midnight Rollover Resolution:**
   - Because `Sensor.TYPE_STEP_COUNTER` emits cumulative steps since the last boot, `ProfilePreferences` persists `lastSensorStepValue: Long`.
   - `StepDeltaProcessor` processes every sensor event:
     - **First Reading (`lastSensorStepValue == -1L`):** Sets `newBaseline = current`, `stepsToAdd = 0`.
     - **Paused State (`isPaused == true`):** Advances `newBaseline = current` and returns `stepsToAdd = 0`. This guarantees steps taken while paused (e.g., in transit) are never added when the user resumes tracking.
     - **Reboot Detected (`current < lastSensorStepValue`):** Recognizes hardware counter reset and sets `stepsToAdd = current`, `newBaseline = current`.
     - **Normal Increment (`current >= lastSensorStepValue`):** Sets `stepsToAdd = current - lastSensorStepValue`, `newBaseline = current`.
   - `StepRepository` queries `LocalDate.now()` and `LocalTime.now().hour` on each batch, seamlessly creating a new `DailyStepEntity` at midnight and bucketing steps into the active `HourlyStepEntity`.
3. **Android 14/15 (`targetSdk 35`) Foreground Service Safety:**
   - Declared as `android:foregroundServiceType="health"` and promoted via `ServiceCompat.startForeground(..., ServiceInfo.FOREGROUND_SERVICE_TYPE_HEALTH)`.
   - Both `BootReceiver` (`ACTION_BOOT_COMPLETED`) and `StepCounterService.onStartCommand()` explicitly verify `ContextCompat.checkSelfPermission(context, Manifest.permission.ACTIVITY_RECOGNITION) == PERMISSION_GRANTED` before starting or promoting the service, preventing `SecurityException` crashes on Android 14/15.

## 5. Physical Calculation Models

Calculations are parameterized using the user's personal profile (stored locally in DataStore):

1. **Step Length ($L$ in meters):**
   - Default based on user height ($H$ in cm):
     $$\text{Female: } L = H \times 0.413 \div 100$$
     $$\text{Male: } L = H \times 0.415 \div 100$$
   - Can be manually overridden in settings.
2. **Distance ($D$ in km):**
   $$D = \frac{\text{Steps} \times L}{1000}$$
3. **Active Calories Burned ($C$ in kcal):**
   - Based on standard walking MET ($\approx 3.5$) and body weight ($W$ in kg):
     $$C = \text{Steps} \times W \times 0.00075$$
4. **Active Time ($T$ in minutes):**
   - Calculated from active walking cadences (estimated at $\approx 100\text{ steps/minute}$).

```kotlin
package com.nordic.stepcounter.domain

object StepCalculationEngine {
    private const val MALE_STRIDE_RATIO = 0.415f
    private const val FEMALE_STRIDE_RATIO = 0.413f
    private const val CALORIE_MET_FACTOR = 0.00075f
    private const val ESTIMATED_STEPS_PER_MINUTE = 100

    fun calculateDefaultStrideLengthMeters(heightCm: Float, isMale: Boolean): Float {
        val ratio = if (isMale) MALE_STRIDE_RATIO else FEMALE_STRIDE_RATIO
        return (heightCm * ratio) / 100f
    }

    fun calculateDistanceKm(steps: Int, strideMeters: Float): Float {
        if (steps <= 0 || strideMeters <= 0f) return 0f
        return (steps * strideMeters) / 1000f
    }

    fun calculateCaloriesKcal(steps: Int, weightKg: Float): Float {
        if (steps <= 0 || weightKg <= 0f) return 0f
        return steps * weightKg * CALORIE_MET_FACTOR
    }

    fun calculateActiveMinutes(steps: Int): Int {
        if (steps <= 0) return 0
        return steps / ESTIMATED_STEPS_PER_MINUTE
    }
}
```

```kotlin
package com.nordic.stepcounter.domain

data class StepDeltaResult(
    val stepsToAdd: Int,
    val newBaselineSensorSteps: Long
)

object StepDeltaProcessor {
    /**
     * Resolves cumulative Sensor.TYPE_STEP_COUNTER values into valid step increments,
     * handling first initialization (-1L), device reboots, and pause states.
     */
    fun computeDelta(
        currentSensorSteps: Long,
        lastSensorSteps: Long,
        isPaused: Boolean
    ): StepDeltaResult {
        if (currentSensorSteps < 0L) {
            return StepDeltaResult(0, lastSensorSteps)
        }

        // First sensor reading ever recorded
        if (lastSensorSteps < 0L) {
            return StepDeltaResult(
                stepsToAdd = 0,
                newBaselineSensorSteps = currentSensorSteps
            )
        }

        // While paused, advance baseline so steps taken during pause are discarded
        if (isPaused) {
            return StepDeltaResult(
                stepsToAdd = 0,
                newBaselineSensorSteps = currentSensorSteps
            )
        }

        // Device rebooted since last reading (hardware counter reset to 0)
        val rawDelta = if (currentSensorSteps < lastSensorSteps) {
            currentSensorSteps
        } else {
            currentSensorSteps - lastSensorSteps
        }

        return StepDeltaResult(
            stepsToAdd = rawDelta.coerceAtLeast(0L).toInt(),
            newBaselineSensorSteps = currentSensorSteps
        )
    }
}
```

Handles sensor step ingestion using `StepDeltaProcessor`, midnight rollover by querying `LocalDate.now()`, and dynamic goal updates:

```kotlin
package com.nordic.stepcounter.data.repository

import com.nordic.stepcounter.data.local.DailyStepEntity
import com.nordic.stepcounter.data.local.HourlyStepEntity
import com.nordic.stepcounter.data.local.StepDao
import com.nordic.stepcounter.data.preferences.ProfilePreferences
import com.nordic.stepcounter.data.preferences.UserProfileState
import com.nordic.stepcounter.domain.StepCalculationEngine
import com.nordic.stepcounter.domain.StepDeltaProcessor
import com.nordic.stepcounter.ui.theme.AppTheme
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.first
import java.time.LocalDate
import java.time.LocalTime

class StepRepository(
    private val stepDao: StepDao,
    private val preferences: ProfilePreferences
) {
    val userProfileFlow: Flow<UserProfileState> = preferences.userProfileFlow

    fun getDailyRecordFlow(date: String): Flow<DailyStepEntity?> =
        stepDao.getDailyRecord(date)

    fun getRecentDailyRecordsFlow(limit: Int): Flow<List<DailyStepEntity>> =
        stepDao.getRecentDailyRecords(limit)

    fun getHourlyRecordsFlow(date: String): Flow<List<HourlyStepEntity>> =
        stepDao.getHourlyRecordsForDate(date)

    suspend fun processHardwareSensorReading(
        rawSensorSteps: Long,
        date: String = LocalDate.now().toString(),
        hour: Int = LocalTime.now().hour
    ): DailyStepEntity {
        val profile = preferences.userProfileFlow.first()
        val deltaResult = StepDeltaProcessor.computeDelta(
            currentSensorSteps = rawSensorSteps,
            lastSensorSteps = profile.lastSensorStepValue,
            isPaused = profile.isPaused
        )

        preferences.updateLastSensorStepValue(deltaResult.newBaselineSensorSteps)

        val existingDaily = stepDao.getDailyRecordOnce(date)
        val newTotalSteps = (existingDaily?.steps ?: 0) + deltaResult.stepsToAdd
        val strideMeters = profile.effectiveStrideMeters

        val updatedDaily = DailyStepEntity(
            date = date,
            steps = newTotalSteps,
            distanceMeters = StepCalculationEngine.calculateDistanceKm(newTotalSteps, strideMeters) * 1000f,
            caloriesKcal = StepCalculationEngine.calculateCaloriesKcal(newTotalSteps, profile.weightKg),
            activeDurationSeconds = StepCalculationEngine.calculateActiveMinutes(newTotalSteps) * 60L,
            goalSteps = profile.dailyGoal
        )
        stepDao.upsertDaily(updatedDaily)

        if (deltaResult.stepsToAdd > 0) {
            val existingHourly = stepDao.getHourlyRecordOnce(date, hour)
            val newHourlySteps = (existingHourly?.steps ?: 0) + deltaResult.stepsToAdd
            stepDao.upsertHourly(
                HourlyStepEntity(date = date, hour = hour, steps = newHourlySteps)
            )
        }

        return updatedDaily
    }

    suspend fun setPaused(paused: Boolean) = preferences.setPaused(paused)

    suspend fun updateDailyGoal(goal: Int) {
        preferences.updateDailyGoal(goal)
        val today = LocalDate.now().toString()
        val existing = stepDao.getDailyRecordOnce(today)
        if (existing != null) {
            stepDao.upsertDaily(existing.copy(goalSteps = goal))
        }
    }

    suspend fun updateProfile(heightCm: Float, weightKg: Float, isMale: Boolean, customStrideMeters: Float) {
        preferences.updateProfile(heightCm, weightKg, isMale, customStrideMeters)
    }

    suspend fun updateAppTheme(theme: AppTheme) {
        preferences.updateAppTheme(theme)
    }
}
```

**Step 2: Implement `StepSensorManager.kt`**
Uses `Sensor.TYPE_STEP_COUNTER` (19) exclusively with a 60-second hardware batching latency (`60_000_000` microseconds) for minimal battery drain:

```kotlin
package com.nordic.stepcounter.service

import android.content.Context
import android.hardware.Sensor
import android.hardware.SensorEvent
import android.hardware.SensorEventListener
import android.hardware.SensorManager

class StepSensorManager(
    context: Context,
    private val onSensorReading: (Long) -> Unit
) : SensorEventListener {

    private val sensorManager = context.getSystemService(Context.SENSOR_SERVICE) as SensorManager
    private val stepCounterSensor: Sensor? = sensorManager.getDefaultSensor(Sensor.TYPE_STEP_COUNTER)

    fun startListening() {
        stepCounterSensor?.let { sensor ->
            // 60 seconds hardware batching in low-power DSP
            sensorManager.registerListener(
                this,
                sensor,
                SensorManager.SENSOR_DELAY_NORMAL,
                60_000_000
            )
        }
    }

    fun stopListening() {
        sensorManager.unregisterListener(this)
    }

    override fun onSensorChanged(event: SensorEvent?) {
        if (event?.sensor?.type == Sensor.TYPE_STEP_COUNTER) {
            val totalStepsSinceBoot = event.values[0].toLong()
            onSensorReading(totalStepsSinceBoot)
        }
    }

    override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) = Unit
}
```

**Step 3: Implement `StepCounterService.kt` with Android 14/15 `FOREGROUND_SERVICE_TYPE_HEALTH` guards**

```kotlin
package com.nordic.stepcounter.service

import android.Manifest
import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.content.pm.ServiceInfo
import android.os.Build
import android.os.IBinder
import androidx.core.app.NotificationCompat
import androidx.core.app.ServiceCompat
import androidx.core.content.ContextCompat
import com.nordic.stepcounter.MainActivity
import com.nordic.stepcounter.data.local.StepDatabase
import com.nordic.stepcounter.data.preferences.ProfilePreferences
import com.nordic.stepcounter.data.repository.StepRepository
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.flow.first
import kotlinx.coroutines.launch
import java.time.LocalDate

class StepCounterService : Service() {

    companion object {
        const val CHANNEL_ID = "step_counter_channel"
        const val NOTIFICATION_ID = 1001
        const val ACTION_TOGGLE_PAUSE = "com.nordic.stepcounter.ACTION_TOGGLE_PAUSE"

        fun hasActivityRecognitionPermission(context: Context): Boolean {
            return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                ContextCompat.checkSelfPermission(
                    context,
                    Manifest.permission.ACTIVITY_RECOGNITION
                ) == PackageManager.PERMISSION_GRANTED
            } else {
                true
            }
        }
    }

    private val serviceScope = CoroutineScope(SupervisorJob() + Dispatchers.IO)
    private lateinit var repository: StepRepository
    private lateinit var sensorManager: StepSensorManager

    override fun onCreate() {
        super.onCreate()
        val db = StepDatabase.getInstance(this)
        val prefs = ProfilePreferences.getInstance(this)
        repository = StepRepository(db.stepDao(), prefs)

        createNotificationChannel()
        sensorManager = StepSensorManager(this) { rawSteps ->
            serviceScope.launch {
                val updated = repository.processHardwareSensorReading(rawSteps)
                val profile = repository.userProfileFlow.first()
                updateNotification(updated.steps, updated.goalSteps, profile.isPaused)
            }
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        // Android 14/15 safety guard: stop immediately if ACTIVITY_RECOGNITION was revoked
        if (!hasActivityRecognitionPermission(this)) {
            stopSelf()
            return START_NOT_STICKY
        }

        if (intent?.action == ACTION_TOGGLE_PAUSE) {
            serviceScope.launch {
                val currentProfile = repository.userProfileFlow.first()
                val newPaused = !currentProfile.isPaused
                repository.setPaused(newPaused)
                val todayRecord = repository.getDailyRecordFlow(LocalDate.now().toString()).first()
                updateNotification(
                    steps = todayRecord?.steps ?: 0,
                    goal = currentProfile.dailyGoal,
                    isPaused = newPaused
                )
            }
        }

        val initialNotification = buildNotification(steps = 0, goal = 6000, isPaused = false)
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.UPSIDE_DOWN_CAKE) {
            ServiceCompat.startForeground(
                this,
                NOTIFICATION_ID,
                initialNotification,
                ServiceInfo.FOREGROUND_SERVICE_TYPE_HEALTH
            )
        } else {
            startForeground(NOTIFICATION_ID, initialNotification)
        }

        sensorManager.startListening()
        return START_STICKY
    }

    override fun onDestroy() {
        sensorManager.stopListening()
        serviceScope.cancel()
        super.onDestroy()
    }

    override fun onBind(intent: Intent?): IBinder? = null

    private fun createNotificationChannel() {
        val channel = NotificationChannel(
            CHANNEL_ID,
            "Step Tracking Service",
            NotificationManager.IMPORTANCE_LOW
        ).apply {
            description = "Shows live daily step count and goal progress"
            setShowBadge(false)
        }
        val manager = getSystemService(NotificationManager::class.java)
        manager.createNotificationChannel(channel)
    }

    private fun updateNotification(steps: Int, goal: Int, isPaused: Boolean) {
        val manager = getSystemService(NotificationManager::class.java)
        manager.notify(NOTIFICATION_ID, buildNotification(steps, goal, isPaused))
    }

    private fun buildNotification(steps: Int, goal: Int, isPaused: Boolean): Notification {
        val openAppIntent = PendingIntent.getActivity(
            this,
            0,
            Intent(this, MainActivity::class.java),
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val togglePauseIntent = PendingIntent.getService(
            this,
            1,
            Intent(this, StepCounterService::class.java).apply { action = ACTION_TOGGLE_PAUSE },
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val percent = if (goal > 0) ((steps * 100) / goal).coerceAtMost(100) else 0
        val statusText = if (isPaused) "Paused • $percent% of goal" else "$percent% of $goal goal"
        val actionLabel = if (isPaused) "Resume" else "Pause"

        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setSmallIcon(android.R.drawable.ic_menu_directions)
            .setContentTitle("$steps STEPS")
            .setContentText(statusText)
            .setOngoing(true)
            .setSilent(true)
            .setContentIntent(openAppIntent)
            .addAction(0, actionLabel, togglePauseIntent)
            .build()
    }
}
```

```kotlin
class BootReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent?) {
        val action = intent?.action ?: return
        if (action == Intent.ACTION_BOOT_COMPLETED || action == Intent.ACTION_MY_PACKAGE_REPLACED) {
            if (StepCounterService.hasActivityRecognitionPermission(context)) {
                val serviceIntent = Intent(context, StepCounterService::class.java)
                ContextCompat.startForegroundService(context, serviceIntent)
            }
        }
    }
}
```

```kotlin
package com.nordic.stepcounter.ui

import androidx.lifecycle.ViewModel
import androidx.lifecycle.ViewModelProvider
import androidx.lifecycle.viewModelScope
import com.nordic.stepcounter.data.local.DailyStepEntity
import com.nordic.stepcounter.data.local.HourlyStepEntity
import com.nordic.stepcounter.data.repository.StepRepository
import com.nordic.stepcounter.domain.StepCalculationEngine
import com.nordic.stepcounter.ui.theme.AppTheme
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch
import java.time.LocalDate

data class StepUiState(
    val todaySteps: Int = 0,
    val dailyGoal: Int = 6000,
    val distanceKm: Float = 0f,
    val caloriesKcal: Float = 0f,
    val activeMinutes: Int = 0,
    val isPaused: Boolean = false,
    val heightCm: Float = 175f,
    val weightKg: Float = 70f,
    val isMale: Boolean = true,
    val customStrideMeters: Float = 0f,
    val effectiveStrideMeters: Float = 0.72f,
    val appTheme: AppTheme = AppTheme.NORDIC_DARK,
    val recentRecords: List<DailyStepEntity> = emptyList(),
    val hourlyToday: List<HourlyStepEntity> = emptyList()
)

class StepViewModel(
    private val repository: StepRepository,
    currentDateProvider: () -> String = { LocalDate.now().toString() }
) : ViewModel() {

    private val todayDate = currentDateProvider()

    val uiState: StateFlow<StepUiState> = combine(
        repository.userProfileFlow,
        repository.getDailyRecordFlow(todayDate),
        repository.getRecentDailyRecordsFlow(30),
        repository.getHourlyRecordsFlow(todayDate)
    ) { profile, todayRecord, recentList, hourlyList ->
        val steps = todayRecord?.steps ?: 0
        val stride = profile.effectiveStrideMeters
        StepUiState(
            todaySteps = steps,
            dailyGoal = profile.dailyGoal,
            distanceKm = StepCalculationEngine.calculateDistanceKm(steps, stride),
            caloriesKcal = StepCalculationEngine.calculateCaloriesKcal(steps, profile.weightKg),
            activeMinutes = StepCalculationEngine.calculateActiveMinutes(steps),
            isPaused = profile.isPaused,
            heightCm = profile.heightCm,
            weightKg = profile.weightKg,
            isMale = profile.isMale,
            customStrideMeters = profile.customStrideLengthMeters,
            effectiveStrideMeters = stride,
            appTheme = profile.appTheme,
            recentRecords = recentList,
            hourlyToday = hourlyList
        )
    }.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5000),
        initialValue = StepUiState()
    )

    fun togglePause() {
        viewModelScope.launch {
            repository.setPaused(!uiState.value.isPaused)
        }
    }

    fun updateGoal(newGoal: Int) {
        viewModelScope.launch {
            repository.updateDailyGoal(newGoal)
        }
    }

    fun updateProfile(heightCm: Float, weightKg: Float, isMale: Boolean, customStrideMeters: Float) {
        viewModelScope.launch {
            repository.updateProfile(heightCm, weightKg, isMale, customStrideMeters)
        }
    }

    fun selectTheme(theme: AppTheme) {
        viewModelScope.launch {
            repository.updateAppTheme(theme)
        }
    }

    class Factory(private val repository: StepRepository) : ViewModelProvider.Factory {
        @Suppress("UNCHECKED_CAST")
        override fun <T : ViewModel> create(modelClass: Class<T>): T {
            return StepViewModel(repository) as T
        }
    }
}
```

```kotlin
                color = palette.trackBackground,

                sweepAngle = 360f,
                useCenter = false,
                style = Stroke(width = strokeWidth, cap = StrokeCap.Round)
            )
            drawArc(
                color = if (isPaused) palette.mutedText else palette.accent,
                startAngle = -90f,

                text = "%,d".format(steps),
                style = MaterialTheme.typography.displayLarge,
```

```kotlin
package com.nordic.stepcounter.ui.components

import androidx.compose.foundation.Canvas
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.foundation.layout.height
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.geometry.CornerRadius
import androidx.compose.ui.geometry.Offset
import androidx.compose.ui.geometry.Size
import androidx.compose.ui.unit.dp
import com.nordic.stepcounter.ui.theme.LocalStepPalette

@Composable
fun SwissBarChart(
    values: List<Int>,
    maxValue: Int,
    modifier: Modifier = Modifier
) {
    val palette = LocalStepPalette.current
    val safeMax = maxValue.coerceAtLeast(1)

    Canvas(
        modifier = modifier
            .fillMaxWidth()
            .height(180.dp)
    ) {
        if (values.isEmpty()) return@Canvas
        val barCount = values.size
        val spacing = 8.dp.toPx()
        val totalSpacing = spacing * (barCount - 1)
        val barWidth = ((size.width - totalSpacing) / barCount).coerceAtLeast(4f)

        values.forEachIndexed { index, value ->
            val ratio = (value.toFloat() / safeMax.toFloat()).coerceIn(0.04f, 1f)
            val barHeight = size.height * ratio
            val x = index * (barWidth + spacing)

            drawRoundRect(
                color = if (value > 0) palette.accent else palette.trackBackground,
                topLeft = Offset(x, y),
                size = Size(barWidth, barHeight),
                cornerRadius = CornerRadius(4.dp.toPx(), 4.dp.toPx())
            )
        }
```

```kotlin
            repository.getRecentDailyRecordsFlow(30),
            repository.getHourlyRecordsFlow(todayDate),

            val dailyGoal = todayRecord?.goalSteps?.takeIf { it > 0 } ?: profile.dailyGoal
```

## Executive Summary

**Nordic Step Counter** is a well-structured, clean-room Android pedometer that achieves its core goals: **zero network permissions**, **zero third-party trackers/ads**, **low-power hardware sensor batching**, and a **distinctive Swiss/Nordic UI** with 8 custom color themes.

| Category | Rating | Summary |
| :--- | :---: | :--- |
| **Privacy & Security** | **9.5 / 10** | No `INTERNET` or GPS permissions; all data stays in app-private storage. Minor `BootReceiver` export flag issue. |
| **Battery & Sensor Design** | **8.5 / 10** | Proper use of `Sensor.TYPE_STEP_COUNTER` (type 19) with 60s hardware FIFO batching (`60_000_000` µs). Needs sensor flush on Pause/Resume. |
| **Architecture & Concurrency** | **7.5 / 10** | Clean UDF pattern with Room + DataStore Flows, but suffers from a `StepRepository` split-brain instantiation and a stale date in `StepViewModel`. |
| **UI / UX & Android 15 Readiness** | **7.0 / 10** | Great typography and palette system, but biometrics cannot be edited in the UI, `HistoryScreen` misrenders sparse days, and status bar icons don't adapt to Light/Dark themes on Android 15. |
| **Testing** | **9.0 / 10** | Comprehensive unit and Robolectric tests across domain, DAO, DataStore, repository, service, and ViewModel layers. |

---

## 1. Key Strengths

1. **Exemplary Hardware Sensor Efficiency (`StepSensorManager.kt`):**
   Exclusively using `Sensor.TYPE_STEP_COUNTER` with `BATCH_LATENCY_US = 60_000_000` (60 seconds) allows the device's low-power Sensor Hub DSP to buffer steps while the AP/CPU sleeps, avoiding the severe battery drain of `TYPE_ACCELEROMETER` or unbatched `TYPE_STEP_DETECTOR` wakeups.
2. **Android 14/15 (`targetSdk 35`) Foreground Service Compliance (`StepCounterService.kt`, `BootReceiver.kt`):**
   Both `BootReceiver` and `StepCounterService` explicitly guard `startForegroundService()` and `startForeground(..., FOREGROUND_SERVICE_TYPE_HEALTH)` behind `hasActivityRecognitionPermission()`, preventing `SecurityException` and `ForegroundServiceStartNotAllowedException` crashes on Android 14+.
3. **Pure, Testable Domain Logic (`StepDeltaProcessor.kt`, `StepCalculationEngine.kt`):**
   Keeping sensor baseline resolution (first-reading initialization, reboot detection, pause handling, `Int.MAX_VALUE` clamping, and `NaN`/`Infinity` guards) in pure Kotlin objects makes the math deterministic and easy to test.
4. **Zero-Boilerplate Theme Engine (`Color.kt`, `Theme.kt`):**
   Providing `StepColorPalette` via `staticCompositionLocalOf` (`LocalStepPalette`) alongside Material 3 `ColorScheme` makes switching between all 8 dark/light themes instantaneous and clean.

## 2. Critical Bugs & Functional Issues (Must Fix)

### 2.1. Midnight Rollover Bug in `StepViewModel` (UI Freezes on Yesterday's Date)
In `StepViewModel.kt` (lines 18–23):
```kotlin
class StepViewModel(
    private val repository: StepRepository,
    currentDateProvider: () -> String = { LocalDate.now().toString() }
) : ViewModel() {

    private val todayDate = currentDateProvider()
```
* **The Bug:** `todayDate` is evaluated **once** when `StepViewModel` is created. If the user keeps the app alive in the background across midnight, `StepCounterService` will start writing steps to the new date (`LocalDate.now()`), while `StepViewModel` continues observing `repository.getDailyRecordFlow(todayDate)` and `getHourlyRecordsFlow(todayDate)` for **yesterday**. The UI will appear frozen until the Activity/process is recreated.
* **Secondary Issue:** In `StepViewModel.init`, `repository.initializeTodayRecord()` is called without passing `todayDate`, bypassing `currentDateProvider`.
* **Fix:** Emit the current date via a `Flow` (or observe `profile.lastRecordedDate` / a periodic date ticker) and use `flatMapLatest` so the ViewModel automatically switches to the new day's Room flows at midnight.

---

### 2.2. Pause/Resume Race Condition with 60-Second Sensor Batching & Notification Desync
Look at how pausing works across `TodayScreen.kt`, `StepViewModel.kt`, `StepCounterService.kt`, and `StepSensorManager.kt`:
1. **Uncounted Steps Before Pause / Counted Steps During Pause:**
   Because `StepSensorManager` batches hardware events in 60-second windows, toggling `isPaused` in DataStore without flushing the hardware sensor FIFO queue causes steps taken *before* pressing Pause (still sitting in the hardware batch buffer) to arrive *after* `isPaused == true` and get discarded. Conversely, when pressing **Resume**, steps taken while paused that haven't been flushed yet arrive *after* `isPaused == false` and get added to the user's daily step count.
2. **Notification Desynchronization:**
   When the user taps **PAUSE TRACKING** on `TodayScreen`, `viewModel.togglePause()` updates DataStore (`repository.setPaused(!currentPaused)`), but **`StepCounterService` is never notified**. Until the next sensor event fires, the ongoing system notification still displays `"X% of 6000 goal"` with a `"Pause"` button even though tracking is paused.
* **Fix:**
  * Either route pause/resume actions through `StepCounterService` (`ACTION_TOGGLE_PAUSE`) or have `StepCounterService` collect `repository.isPausedFlow` to immediately update the notification and call `sensorManager.flush(this)` before applying the state transition.

---

### 2.3. `StepRepository` Singleton Split-Brain
In `StepCounterService.kt` (line 53), the service uses the singleton:
```kotlin
repository = StepRepository.getInstance(this)
```
However, in `MainActivity.kt` (lines 40–44), the Activity instantiates a **separate** `StepRepository`:
```kotlin
private val viewModel: StepViewModel by viewModels {
    val db = StepDatabase.getInstance(applicationContext)
    val prefs = ProfilePreferences.getInstance(applicationContext)
    StepViewModel.Factory(StepRepository(db.stepDao(), prefs))
}
```
* **The Bug:** `StepRepository` holds an instance-level `private val sensorMutex = Mutex()`. Having two separate instances in the same process means `MainActivity` and `StepCounterService` do not share the repository instance. Furthermore, `initializeTodayRecord()` and `updateDailyGoal()` in `StepRepository` perform read-then-upsert operations on `daily_steps` outside `sensorMutex`, which can race with `processHardwareSensorReading()`.
* **Fix:** Use `StepViewModel.provideFactory(applicationContext)` (which already exists in `StepViewModel.Companion` and calls `StepRepository.getInstance(context)`) in `MainActivity`, and wrap `initializeTodayRecord()` and `updateDailyGoal()` inside `sensorMutex.withLock`.

---

### 2.4. Missing Biometric Calibration UI in `ProfileScreen.kt`
`ProfilePreferences`, `StepRepository`, and `StepViewModel` all implement `updatePhysicalProfile(heightCm, weightKg, isMale, customStrideMeters)`. However, `ProfileScreen.kt` (lines 72–82) only renders static text:
```kotlin
Text(
    text = "Height: ${state.heightCm.toInt()} cm • Weight: ${state.weightKg.toInt()} kg • Stride: ${"%.2f".format(state.effectiveStrideMeters)} m",
    style = MaterialTheme.typography.titleMedium,
    color = palette.mutedText
)
```
* **The Impact:** Users have no way to change their height, weight, biological sex, or custom stride length anywhere in the app—everyone is locked to the defaults (`175 cm`, `70 kg`, `Male`, `0.73 m` stride).
* **Fix:** Add an edit dialog or inline sliders/steppers in `ProfileScreen` wired to `viewModel::updatePhysicalProfile`.

## 3. UI/UX & Android 15 (`targetSdk 35`) Issues

### 3.1. `HistoryScreen` Sparse-Data Chart Bug & Unfiltered Summary Cards
In `HistoryScreen.kt` (lines 30–42):
```kotlin
val chartValues = when (selectedTab) {
    0 -> (0..23).map { hour -> state.hourlyToday.firstOrNull { it.hour == hour }?.steps ?: 0 }
    1 -> state.recentRecords.take(7).reversed().map { it.steps }
    else -> state.recentRecords.take(30).reversed().map { it.steps }

```
1. **Single Giant Bar on New Installs:** On Day 1, `state.recentRecords` contains only 1 item. Selecting `7 DAYS` or `30 DAYS` passes a 1-element list to `SwissBarChart`, which divides the canvas width by `barCount = 1` and draws a single full-width block across the entire screen.
2. **Skipped Calendar Days:** If the device was off for a day, `recentRecords.take(7)` takes the last 7 database rows rather than the last 7 calendar dates, shifting the timeline without showing `0` for missed days.
   * **Fix:** Generate the exact date range `(6 downTo 0).map { LocalDate.now().minusDays(it.toLong()).toString() }` (and `29 downTo 0` for 30 days) and lookup each date in `state.recentRecords`, defaulting to `0`.

### 3.2. Android 15 Edge-to-Edge Status Bar Contrast
Because `targetSdk = 35` enforces edge-to-edge on Android 15, switching between the 4 Dark themes (`NORDIC_DARK`, `GRUVBOX_DARK`, etc.) and 4 Light themes (`NORDIC_LIGHT`, `GRUVBOX_LIGHT`, etc.) does not update the system status bar or navigation bar icon luminance. On light themes like `NORDIC_LIGHT` (`#F8F9FA`), light status bar icons become unreadable.
* **Fix:** Inside `NordicStepTheme` (in `Theme.kt`), update `WindowCompat.getInsetsController(window, view).isAppearanceLightStatusBars = !appTheme.isDark`.

### 3.3. State Loss & Repeated Permission Launches on Configuration Change
In `MainActivity.kt` and `HistoryScreen.kt`:
* `var currentTab by remember { mutableIntStateOf(0) }` (`MainActivity.kt:50`) and `var selectedTab by remember { mutableIntStateOf(0) }` (`HistoryScreen.kt:28`) use `remember` instead of `rememberSaveable`. Rotating the device resets navigation back to the `TODAY` tab and resets the history filter.
* `LaunchedEffect(Unit)` in `MainActivity.kt:58` re-triggers `permissionLauncher.launch(...)` on every screen rotation.

### 3.4. Zombie Foreground Service When Hardware Sensor Is Missing
In `StepCounterService.kt` (line 115):
```kotlin
sensorManager.startListening()
```
If a device lacks `Sensor.TYPE_STEP_COUNTER` (e.g., emulators or sideloaded APKs on unsupported devices), `sensorManager.startListening()` returns `false`, yet `StepCounterService` remains in the foreground indefinitely showing `0 STEPS`. If `!sensorManager.isSensorAvailable`, the service should stop itself cleanly and update `_isTrackingActive` in the UI.

## 4. Build, Code Hygiene & Security Improvements

1. **Drop `androidx.compose.material:material-icons-extended` (`libs.versions.toml` line 25):**
   The project includes the full `material-icons-extended` artifact solely for 3 icons (`DirectionsWalk`, `BarChart`, `Person`). In debug builds (and unshrunk configurations), `material-icons-extended` adds ~30 MB of bytecode/resources and slows down compilation. Replace those 3 icons with lightweight vector drawables (`res/drawable/ic_*.xml`) and remove the dependency.
2. **Remove Unused `kotlin-serialization` Plugin (`app/build.gradle.kts` line 4):**
   `alias(libs.plugins.kotlin.serialization)` is applied in Gradle, but `kotlinx.serialization` is not used anywhere in the project.
3. **Set `android:exported="false"` on `BootReceiver` (`AndroidManifest.xml` line 47):**
   `BootReceiver` only listens to system broadcasts (`BOOT_COMPLETED` and `MY_PACKAGE_REPLACED`). Setting `android:exported="false"` prevents other apps on the device from sending an explicit spoofed `BOOT_COMPLETED` intent to `BootReceiver`.
4. **Consolidate Duplicate Alias Methods & Properties:**
   Several classes define duplicate pass-through aliases that add noise to the public API:
   * `UserProfile`: `effectiveStrideLengthMeters` vs `effectiveStrideMeters`
   * `ProfilePreferences`: `updatePhysicalProfile` vs `updateProfile`, `updateLastSensorReading` vs `updateLastSensorStepValue`
   * `StepViewModel`: `updateDailyGoal` vs `updateGoal`, `updatePhysicalProfile` vs `updateProfile`, `updateAppTheme` vs `selectTheme`
   * `StepUiState`: `appTheme` vs `currentTheme`
   * `Theme.kt`: `NordicStepTheme` vs `NordicStepCounterTheme`
5. **Reset `StepRepository` Singleton in `StepServiceTest`:**
   `StepCounterService` calls `StepRepository.getInstance(this)`, which caches a static `INSTANCE` in `StepRepository.Companion`. Add `StepRepository.resetInstance()` in `StepServiceTest.tearDown()` so test runs cannot leak repository state across test cases.
