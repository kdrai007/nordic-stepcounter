import org.junit.Assert.assertEquals

class ScreensUiTest {

    @get:Rule
    val composeRule = createComposeRule()

    @Test
    fun todayScreen_displaysMetricsSemanticsAndDisablesPauseWithoutPermission() {
        var permissionRequested = false
        var downtimeDismissed = false
        val state = StepUiState(
            isLoaded = true,
            todaySteps = 5432,
            dailyGoal = 10000,
            distanceKm = 4.05f,
            caloriesKcal = 240f,
            activeMinutes = 54,
            downtimeGapFromDate = "2026-11-08",
            downtimeGapToDate = "2026-11-10"
        )

        composeRule.setContent {
            NordicStepTheme(appTheme = AppTheme.NORDIC_DARK) {
                TodayScreen(
                    state = state,
                    onTogglePause = {},
                    hasPermission = false,
                    hasNotificationPermission = true,
                    hasStepSensor = true,
                    onRequestPermission = { permissionRequested = true },
                    onDismissDowntimeNotice = { downtimeDismissed = true }
                )
            }
        }

        composeRule.onNodeWithContentDescription("5,432 of 10,000 steps, 54 percent of goal, tracking active")
            .assertIsDisplayed()
        composeRule.onNodeWithText("PAUSE TRACKING").assertIsNotEnabled()

        composeRule.onNodeWithText("GRANT").performClick()
        assertTrue(permissionRequested)

        composeRule.onNodeWithText("OK").performClick()
        assertTrue(downtimeDismissed)
    }

    @Test
    fun historyScreen_switchesTabsAndShowsAllTimePersonalBest() {
        val state = StepUiState(
            isLoaded = true,
            todaySteps = 2400,
            dailyGoal = 8000,
            personalBestSteps = 18500,
            earliestRecordDate = "2026-11-01",
            currentDate = "2026-11-10",
            hourlyToday = listOf(HourlyStepEntity("2026-11-10", 9, 2400)),
            recentRecords = listOf(
                DailyStepEntity("2026-11-09", 8000, 5800f, 300f, 4800L, 8000),
                DailyStepEntity("2026-11-10", 2400, 1700f, 100f, 1440L, 8000)
            )
        )

        composeRule.setContent {
            NordicStepTheme(appTheme = AppTheme.NORDIC_LIGHT) {
                HistoryScreen(state = state)
            }
        }

        composeRule.onNodeWithText("HOURLY AVG").assertIsDisplayed()
        composeRule.onNodeWithText("18,500").assertIsDisplayed()
        composeRule.onNodeWithText("PERSONAL BEST").assertIsDisplayed()

        composeRule.onNodeWithText("7 DAYS").performClick()
        composeRule.onNodeWithText("DAILY AVG").assertIsDisplayed()
    }

    @Test
    fun profileScreen_allowsCustomGoalAndThemeSelection() {
        var selectedGoal = 6000
        var selectedTheme = AppTheme.NORDIC_DARK
        val state = StepUiState(
            isLoaded = true,
            dailyGoal = 6000,
            appTheme = AppTheme.NORDIC_DARK
        )

        composeRule.setContent {
            NordicStepTheme(appTheme = state.appTheme) {
                ProfileScreen(
                    state = state,
                    onSelectGoal = { selectedGoal = it },
                    onSelectTheme = { selectedTheme = it }
                )
            }
        }

        composeRule.onNodeWithText("CUSTOM").performClick()
        composeRule.onNodeWithText("6000").performTextReplacement("7500")
        composeRule.onNodeWithText("SAVE").performClick()
        assertEquals(7500, selectedGoal)

        composeRule.onNodeWithText("Tokyo Night Light").performClick()
        assertEquals(AppTheme.TOKYONIGHT_LIGHT, selectedTheme)
    }
}

        val started = startStepForegroundServiceSafely(intent)

@Dao
interface StepDao {
    @Query("SELECT * FROM daily_steps WHERE date = :date LIMIT 1")
    fun getDailyRecord(date: String): Flow<DailyStepEntity?>

    @Query("SELECT * FROM daily_steps WHERE date = :date LIMIT 1")
    suspend fun getDailyRecordDirect(date: String): DailyStepEntity?

    @Query("SELECT * FROM daily_steps ORDER BY date DESC LIMIT :limit")
    fun getRecentDailyRecords(limit: Int): Flow<List<DailyStepEntity>>

    fun getDailyRecordsInRange(startDate: String, endDate: String): Flow<List<DailyStepEntity>>

    @Query("SELECT MAX(steps) FROM daily_steps WHERE date <= :maxDate")
    fun getBestDaySteps(maxDate: String): Flow<Int?>

    @Query("SELECT MIN(date) FROM daily_steps WHERE date <= :maxDate")
    fun getEarliestRecordDate(maxDate: String): Flow<String?>

    @Query("SELECT * FROM hourly_steps WHERE date = :date ORDER BY hour ASC")
    fun getHourlyRecordsForDate(date: String): Flow<List<HourlyStepEntity>>

    @Query("SELECT * FROM hourly_steps WHERE date = :date AND hour = :hour LIMIT 1")
    suspend fun getHourlyRecordDirect(date: String, hour: Int): HourlyStepEntity?

    @Query("SELECT * FROM sensor_state WHERE id = ${SensorStateEntity.SINGLETON_ID} LIMIT 1")
    fun getSensorStateFlow(): Flow<SensorStateEntity?>

    @Query("SELECT * FROM sensor_state WHERE id = ${SensorStateEntity.SINGLETON_ID} LIMIT 1")
    suspend fun getSensorStateDirect(): SensorStateEntity?

    @Upsert
    suspend fun upsertSensorState(entity: SensorStateEntity)

    @Upsert
    suspend fun upsertDaily(entity: DailyStepEntity)

    @Upsert
    suspend fun upsertHourly(entity: HourlyStepEntity)

    @Transaction
    suspend fun upsertDailyAndHourly(daily: DailyStepEntity, hourly: HourlyStepEntity?) {
        upsertDaily(daily)
        if (hourly != null) {
            upsertHourly(hourly)
        }
    }

    @Transaction
    suspend fun persistSensorUpdate(
        sensorState: SensorStateEntity,
        daily: DailyStepEntity,
        hourly: HourlyStepEntity?
    ) {
        upsertSensorState(sensorState)
        upsertDaily(daily)
        if (hourly != null) {
            upsertHourly(hourly)
        }
    }

    @Query("UPDATE daily_steps SET goalSteps = :goal WHERE date = :date")
    suspend fun updateDailyGoal(date: String, goal: Int): Int

    @Query("SELECT SUM(steps) FROM daily_steps")
    suspend fun getTotalLifetimeSteps(): Long?
}

    suspend fun processHardwareSensorReading(
        rawSensorSteps: Long,
        date: String? = null,
        hour: Int? = null,
        eventTimestampNanos: Long = 0L
    ): DailyStepEntity = sensorMutex.withLock {
        val profile = preferences.userProfileFlow.first()
        val sensorState = getOrMigrateSensorStateLocked()

        val nowMillis = currentTimeMillisProvider()
        val eventTimeMillis = resolveEventTimeMillis(
            eventTimestampNanos = eventTimestampNanos,
            nowMillis = nowMillis,
            elapsedRealtimeNanos = elapsedRealtimeNanosProvider()
        )
        val zonedDateTime = Instant.ofEpochMilli(eventTimeMillis).atZone(zoneIdProvider())
        val resolvedDate = date ?: zonedDateTime.toLocalDate().toString()
        val resolvedHour = (hour ?: zonedDateTime.hour).coerceIn(0, 23)

        if (rawSensorSteps < 0L) {
            return@withLock stepDao.getDailyRecordDirect(resolvedDate) ?: DailyStepEntity(
                date = resolvedDate,
                steps = 0,
                distanceMeters = 0f,
                caloriesKcal = 0f,
                activeDurationSeconds = 0L,
                goalSteps = profile.dailyGoal
            )
        }

        val currentInstallId = installationIdProvider()
        val isCrossDeviceRestore = sensorState.installationId.isNotEmpty() &&
            sensorState.installationId != currentInstallId

        val resolvedLocalDate = runCatching { LocalDate.parse(resolvedDate) }.getOrNull()
        val lastRecordedLocalDate = sensorState.lastRecordedDate
            .takeIf { it.isNotBlank() }
            ?.let { runCatching { LocalDate.parse(it) }.getOrNull() }
        val daysBetween = if (resolvedLocalDate != null && lastRecordedLocalDate != null) {
            ChronoUnit.DAYS.between(lastRecordedLocalDate, resolvedLocalDate)
        } else {
            0L
        }
        val isMultiDayGap = daysBetween > 1L

        val currentBootCount = bootCountProvider()
        val deltaResult = StepDeltaProcessor.computeDelta(
            currentSensorSteps = rawSensorSteps,
            lastSensorSteps = sensorState.lastSensorStepValue,
            isPaused = sensorState.isPaused,
            currentBootCount = currentBootCount,
            lastBootCount = sensorState.lastBootCount,
            pendingResumeBaseline = sensorState.pendingResumeBaseline,
            forceBaselineReset = isCrossDeviceRestore || isMultiDayGap
        )

        val shouldRecordDowntimeNotice = isMultiDayGap &&
            !isCrossDeviceRestore &&
            !sensorState.isPaused &&
            sensorState.lastSensorStepValue >= 0L &&
            rawSensorSteps != sensorState.lastSensorStepValue

        val nextGapFrom = if (shouldRecordDowntimeNotice) {
            sensorState.lastRecordedDate
        } else {
            sensorState.downtimeGapFromDate
        }
        val nextGapTo = if (shouldRecordDowntimeNotice) {
            resolvedDate
        } else {
            sensorState.downtimeGapToDate
        }

        val updatedSensorState = sensorState.copy(
            lastSensorStepValue = deltaResult.newBaselineSensorSteps,
            lastBootCount = deltaResult.newBootCount,
            lastRecordedDate = resolvedDate,
            lastEventTimestampMs = eventTimeMillis,
            pendingResumeBaseline = deltaResult.newPendingResumeBaseline,
            installationId = currentInstallId,
            downtimeGapFromDate = nextGapFrom,
            downtimeGapToDate = nextGapTo
        )

        val existingDaily = stepDao.getDailyRecordDirect(resolvedDate)
        val baseSteps = existingDaily?.steps ?: 0
        val newTotalSteps = (baseSteps.toLong() + deltaResult.stepsToAdd)
            .coerceAtMost(Int.MAX_VALUE.toLong()).toInt()

        val strideMeters = profile.effectiveStrideLengthMeters
        val distanceMeters = StepCalculationEngine.calculateDistanceKm(newTotalSteps, strideMeters) * 1000f
        val caloriesKcal = StepCalculationEngine.calculateCaloriesKcal(newTotalSteps, profile.weightKg, strideMeters)
        val activeDurationSeconds = StepCalculationEngine.calculateActiveSeconds(newTotalSteps)

        val updatedDaily = DailyStepEntity(
            date = resolvedDate,
            steps = newTotalSteps,
            distanceMeters = distanceMeters,
            caloriesKcal = caloriesKcal,
            activeDurationSeconds = activeDurationSeconds,
            goalSteps = profile.dailyGoal
        )

        val hourlyEntity = if (deltaResult.stepsToAdd > 0) {
            val existingHourly = stepDao.getHourlyRecordDirect(resolvedDate, resolvedHour)
            val newHourlySteps = ((existingHourly?.steps ?: 0).toLong() + deltaResult.stepsToAdd)
                .coerceAtMost(Int.MAX_VALUE.toLong()).toInt()
            HourlyStepEntity(
                date = resolvedDate,
                hour = resolvedHour,
                steps = newHourlySteps
            )
        } else {
            null
        }

        stepDao.persistSensorUpdate(updatedSensorState, updatedDaily, hourlyEntity)

        return@withLock updatedDaily
    }

    suspend fun setPaused(
        isPaused: Boolean,
        pendingResumeBaseline: Boolean = false
    ) = sensorMutex.withLock {
        val current = getOrMigrateSensorStateLocked()
        val nextPending = if (isPaused) false else pendingResumeBaseline
        stepDao.upsertSensorState(
            current.copy(
                isPaused = isPaused,
                pendingResumeBaseline = nextPending,
                installationId = current.installationId.ifEmpty { installationIdProvider() }
            )
        )
    }

    suspend fun dismissDowntimeNotice() = sensorMutex.withLock {
        val current = getOrMigrateSensorStateLocked()
        if (current.downtimeGapFromDate.isNotEmpty() || current.downtimeGapToDate.isNotEmpty()) {
            stepDao.upsertSensorState(
                current.copy(
                    downtimeGapFromDate = "",
                    downtimeGapToDate = ""
                )
            )
        }
    }

    suspend fun updateDailyGoal(goal: Int) = sensorMutex.withLock {
        val safeGoal = goal.coerceIn(1_000, 100_000)
        preferences.updateDailyGoal(safeGoal)
        val today = LocalDate.now(zoneIdProvider()).toString()
        val affected = stepDao.updateDailyGoal(today, safeGoal)
        if (affected == 0) {
            val existing = stepDao.getDailyRecordDirect(today)
            if (existing == null) {
                stepDao.upsertDaily(
                    DailyStepEntity(
                        date = today,
                        steps = 0,
                        distanceMeters = 0f,
                        caloriesKcal = 0f,
                        activeDurationSeconds = 0L,
                        goalSteps = safeGoal
                    )
                )
            }
        }
    }

    suspend fun updatePhysicalProfile(
        heightCm: Float,
        weightKg: Float,
        isMale: Boolean,
        customStrideLengthMeters: Float = 0.0f
    ) = sensorMutex.withLock {
        val currentProfile = preferences.userProfileFlow.first()
        val safeHeight = if (heightCm.isFinite()) heightCm.coerceIn(100f, 250f) else currentProfile.heightCm
        val safeWeight = if (weightKg.isFinite()) weightKg.coerceIn(30f, 250f) else currentProfile.weightKg
        val safeCustomStride = if (customStrideLengthMeters.isFinite() && customStrideLengthMeters > 0f) {
            customStrideLengthMeters.coerceIn(0.30f, 1.50f)
        } else {
            0.0f
        }
        preferences.updatePhysicalProfile(safeHeight, safeWeight, isMale, safeCustomStride)

        val today = LocalDate.now(zoneIdProvider()).toString()
        val todayRecord = stepDao.getDailyRecordDirect(today)
        if (todayRecord != null) {
            val strideMeters = if (safeCustomStride > 0f) {
                safeCustomStride
            } else {
                StepCalculationEngine.calculateDefaultStrideLengthMeters(safeHeight, isMale)
            }
            val distanceMeters = StepCalculationEngine.calculateDistanceKm(todayRecord.steps, strideMeters) * 1000f
            val caloriesKcal = StepCalculationEngine.calculateCaloriesKcal(todayRecord.steps, safeWeight, strideMeters)
            val activeDurationSeconds = StepCalculationEngine.calculateActiveSeconds(todayRecord.steps)
            stepDao.upsertDaily(
                todayRecord.copy(
                    distanceMeters = distanceMeters,
                    caloriesKcal = caloriesKcal,
                    activeDurationSeconds = activeDurationSeconds
                )
            )
        }
    }

    suspend fun updateAppTheme(theme: AppTheme) {
        preferences.updateAppTheme(theme)
    }

    private suspend fun getOrMigrateSensorStateLocked(): SensorStateEntity {
        val existing = stepDao.getSensorStateDirect()
        if (existing != null) return existing

        val installId = installationIdProvider()
        val legacy = preferences.consumeLegacySensorState()
        val initial = if (legacy != null) {
            SensorStateEntity(
                lastSensorStepValue = legacy.lastSensorStepValue,
                lastBootCount = -1,
                lastRecordedDate = legacy.lastRecordedDate,
                lastEventTimestampMs = 0L,
                isPaused = legacy.isPaused,
                pendingResumeBaseline = false,
                installationId = installId
            )
        } else {
            SensorStateEntity(installationId = installId)
        }
        stepDao.upsertSensorState(initial)
        return initial
    }

        private const val DEFAULT_INSTALLATION_ID = "local"
        private const val INSTALLATION_ID_FILENAME = "nordic_sensor_install_id"
        private const val MAX_FUTURE_TIMESTAMP_SKEW_NANOS = 60_000_000_000L
        private const val MAX_PAST_TIMESTAMP_AGE_NANOS = 30L * 24L * 3600L * 1_000_000_000L

        @Volatile
        private var INSTANCE: StepRepository? = null

        fun resolveEventTimeMillis(
            eventTimestampNanos: Long,
            nowMillis: Long,
            elapsedRealtimeNanos: Long
        ): Long {
            if (eventTimestampNanos <= 0L || elapsedRealtimeNanos <= 0L) {
                return nowMillis
            }
            val deltaNanos = elapsedRealtimeNanos - eventTimestampNanos
            if (deltaNanos < -MAX_FUTURE_TIMESTAMP_SKEW_NANOS || deltaNanos > MAX_PAST_TIMESTAMP_AGE_NANOS) {
                return nowMillis
            }
            return nowMillis - (deltaNanos / 1_000_000L)
        }

        private fun readBootCount(context: Context): Int {
            return runCatching {
                Settings.Global.getInt(context.contentResolver, Settings.Global.BOOT_COUNT, -1)
            }.getOrDefault(-1)
        }

        private fun readOrCreateInstallationId(context: Context): String {
            return runCatching {
                val file = File(context.noBackupFilesDir, INSTALLATION_ID_FILENAME)
                if (file.exists()) {
                    val existing = file.readText().trim()
                    if (existing.isNotEmpty()) return@runCatching existing
                }
                val created = UUID.randomUUID().toString()
                file.parentFile?.mkdirs()
                file.writeText(created)
                created
            }.getOrDefault(DEFAULT_INSTALLATION_ID)
        }

        fun getInstance(context: Context): StepRepository {
            val appContext = context.applicationContext
            return INSTANCE ?: synchronized(this) {
                INSTANCE ?: StepRepository(
                    stepDao = StepDatabase.getInstance(appContext).stepDao(),
                    preferences = ProfilePreferences.getInstance(appContext),
                    bootCountProvider = { readBootCount(appContext) },
                    installationIdProvider = { readOrCreateInstallationId(appContext) }
                ).also { INSTANCE = it }
            }
        }

        fun getInstance(stepDao: StepDao, preferences: ProfilePreferences): StepRepository {
            return INSTANCE ?: synchronized(this) {
                INSTANCE ?: StepRepository(stepDao, preferences).also { INSTANCE = it }
            }
        }

        internal fun setInstanceForTesting(repository: StepRepository) {
            synchronized(this) {
                INSTANCE = repository
            }
        }

        fun resetInstance() {
            synchronized(this) {
                INSTANCE = null
            }
        }
    }

package com.nordic.stepcounter.domain

/**
 * Pure domain calculation engine for physical fitness metrics:
 * Stride length, distance, net walking calories, and active duration.
 */
object StepCalculationEngine {
    private const val MALE_STRIDE_RATIO = 0.415f
    private const val FEMALE_STRIDE_RATIO = 0.413f
    private const val NET_WALKING_KCAL_PER_KG_PER_KM = 0.60f
    private const val ESTIMATED_STEPS_PER_MINUTE = 100

    fun calculateDefaultStrideLengthMeters(heightCm: Float, isMale: Boolean): Float {
        if (!heightCm.isFinite() || heightCm <= 0f) return 0f
        val ratio = if (isMale) MALE_STRIDE_RATIO else FEMALE_STRIDE_RATIO
        return (heightCm * ratio) / 100f
    }

    fun calculateDistanceKm(steps: Int, strideMeters: Float): Float {
        if (steps <= 0 || !strideMeters.isFinite() || strideMeters <= 0f) return 0f
        return (steps * strideMeters) / 1000f
    }

    /**
     * Estimates net active walking expenditure in kcal from distance covered and body mass.
     */
    fun calculateCaloriesKcal(steps: Int, weightKg: Float, strideMeters: Float): Float {
        if (steps <= 0 || !weightKg.isFinite() || weightKg <= 0f) return 0f
        val distanceKm = calculateDistanceKm(steps, strideMeters)
        if (distanceKm <= 0f) return 0f
        return distanceKm * weightKg * NET_WALKING_KCAL_PER_KG_PER_KM
    }

    fun calculateActiveSeconds(steps: Int): Long {
        if (steps <= 0) return 0L
        return (steps.toLong() * 60L) / ESTIMATED_STEPS_PER_MINUTE
    }

    fun calculateActiveMinutes(steps: Int): Int {
        if (steps <= 0) return 0
        return steps / ESTIMATED_STEPS_PER_MINUTE
    }
}

package com.nordic.stepcounter.domain

data class StepDeltaResult(
    val stepsToAdd: Int,
    val newBaselineSensorSteps: Long,
    val newBootCount: Int = -1,
    val newPendingResumeBaseline: Boolean = false
)

/**
 * Resolves cumulative Sensor.TYPE_STEP_COUNTER values into valid step increments,
 * handling first initialization (-1L), authoritative OS boot-count reboots,
 * stale/out-of-order regressions, pause states, and post-resume baseline sync.
 */
object StepDeltaProcessor {

    private const val SENSOR_HUB_RESET_HIGH_WATERMARK = 1_000L
    private const val SENSOR_HUB_RESET_LOW_THRESHOLD = 50L

    fun computeDelta(
        currentSensorSteps: Long,
        lastSensorSteps: Long,
        isPaused: Boolean,
        currentBootCount: Int = -1,
        lastBootCount: Int = -1,
        pendingResumeBaseline: Boolean = false,
        forceBaselineReset: Boolean = false
    ): StepDeltaResult {
        if (currentSensorSteps < 0L) {
            return StepDeltaResult(
                stepsToAdd = 0,
                newBaselineSensorSteps = lastSensorSteps,
                newBootCount = lastBootCount,
                newPendingResumeBaseline = pendingResumeBaseline
            )
        }

        val resolvedBootCount = if (currentBootCount >= 0) currentBootCount else lastBootCount

        // First sensor reading ever recorded or explicit baseline reset (e.g., cross-device restore / multi-day gap)
        if (forceBaselineReset || lastSensorSteps < 0L) {
            return StepDeltaResult(
                stepsToAdd = 0,
                newBaselineSensorSteps = currentSensorSteps,
                newBootCount = resolvedBootCount,
                newPendingResumeBaseline = false
            )
        }

        // While paused, advance baseline so steps taken during pause are discarded
        if (isPaused) {
            return StepDeltaResult(
                stepsToAdd = 0,
                newBaselineSensorSteps = currentSensorSteps,
                newBootCount = resolvedBootCount,
                newPendingResumeBaseline = false
            )
        }

        // Cold-start resume sync: absorb the first sensor reading after resuming as the new baseline
        if (pendingResumeBaseline) {
            return StepDeltaResult(
                stepsToAdd = 0,
                newBaselineSensorSteps = currentSensorSteps,
                newBootCount = resolvedBootCount,
                newPendingResumeBaseline = false
            )
        }

        // Authoritative boot count comparison when both current and stored boot counts are valid
        if (currentBootCount >= 0 && lastBootCount >= 0) {
            if (currentBootCount != lastBootCount) {
                // Real device reboot occurred since last reading
                return StepDeltaResult(
                    stepsToAdd = currentSensorSteps.coerceIn(0L, Int.MAX_VALUE.toLong()).toInt(),
                    newBaselineSensorSteps = currentSensorSteps,
                    newBootCount = currentBootCount,
                    newPendingResumeBaseline = false
                )
            }

            // Same boot session
            if (currentSensorSteps >= lastSensorSteps) {
                val rawDelta = currentSensorSteps - lastSensorSteps
                return StepDeltaResult(
                    stepsToAdd = rawDelta.coerceIn(0L, Int.MAX_VALUE.toLong()).toInt(),
                    newBaselineSensorSteps = currentSensorSteps,
                    newBootCount = currentBootCount,
                    newPendingResumeBaseline = false
                )
            }

            // Regression within the same boot session: either stale reading or rare sensor-hub crash reset near 0
            val isLikelySensorHubReset =
                lastSensorSteps >= SENSOR_HUB_RESET_HIGH_WATERMARK &&
                    currentSensorSteps <= SENSOR_HUB_RESET_LOW_THRESHOLD
            val nextBaseline = if (isLikelySensorHubReset) currentSensorSteps else lastSensorSteps
            return StepDeltaResult(
                stepsToAdd = 0,
                newBaselineSensorSteps = nextBaseline,
                newBootCount = currentBootCount,
                newPendingResumeBaseline = false
            )
        }

        // Fallback when boot count is unavailable (legacy data or test environments)
        val rawDelta = if (currentSensorSteps < lastSensorSteps) {
            currentSensorSteps
        } else {
            currentSensorSteps - lastSensorSteps
        }

        return StepDeltaResult(
            stepsToAdd = rawDelta.coerceIn(0L, Int.MAX_VALUE.toLong()).toInt(),
            newBaselineSensorSteps = currentSensorSteps,
            newBootCount = resolvedBootCount,
            newPendingResumeBaseline = false
        )
    }
}

import android.content.BroadcastReceiver

/**
 * BroadcastReceiver responsible for restarting the [StepCounterService] upon device reboot
 * or app package update. Verifies ACTIVITY_RECOGNITION runtime permission and sensor presence,
 * and guards Foreground Service startup against Android 14/15 background restrictions.
 */
class BootReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent?) {
        val action = intent?.action ?: return
        if (action == Intent.ACTION_BOOT_COMPLETED || action == Intent.ACTION_MY_PACKAGE_REPLACED) {
            if (StepCounterService.hasActivityRecognitionPermission(context) &&
                StepCounterService.hasStepCounterSensor(context)
            ) {
                val serviceIntent = Intent(context, StepCounterService::class.java).apply {
                    this.action = StepCounterService.ACTION_START
                }
                try {
                    ContextCompat.startForegroundService(context, serviceIntent)
                } catch (_: SecurityException) {
                    // Permission revoked before service startup
                } catch (_: IllegalStateException) {
                    // ForegroundServiceStartNotAllowedException on Android 12+ background restriction
                }
            }
        }
    }
}

    internal fun restartActorWithCurrentScope() {
        actorJob?.cancel()
        profileObserverJob?.cancel()
        actorJob = null
        profileObserverJob = null
        startActorIfNeeded()
    }

    private fun startActorIfNeeded() {
        if (actorJob?.isActive == true) return
        actorJob = serviceScope.launch {
            while (true) {
                val command = if (deferredCommands.isNotEmpty()) {
                    deferredCommands.removeFirst()
                } else {
                    commandChannel.receiveCatching().getOrNull() ?: break
                }
                handleCommand(command)
            }
        }
        if (profileObserverJob?.isActive != true) {
            profileObserverJob = serviceScope.launch {
                combine(
                    repository.isPausedFlow,
                    repository.dailyGoalFlow
                ) { isPaused, dailyGoal -> isPaused to dailyGoal }
                    .distinctUntilChanged()
                    .collect {
                        if (isForegroundStarted) {
                            updateNotificationForCurrentState()
                        }
                    }
            }
        }
    }

    private suspend fun handleCommand(command: ServiceCommand) {
        when (command) {
            ServiceCommand.Initialize -> {
                repository.initializeTodayRecord(currentDateProvider())
                updateNotificationForCurrentState()
            }
            is ServiceCommand.SensorReading -> {
                processSingleReading(command.reading)
            }
            ServiceCommand.Flush -> {
                if (::sensorManager.isInitialized) {
                    sensorManager.flushAndWait()
                }
                drainBufferedSensorReadings()
                updateNotificationForCurrentState()
            }
            is ServiceCommand.SetForegroundMode -> {
                if (::sensorManager.isInitialized) {
                    sensorManager.setForegroundMode(command.isForeground)
                }
            }
            is ServiceCommand.SetPaused -> {
                applyPauseStateChange(targetPaused = command.isPaused)
            }
            ServiceCommand.TogglePause -> {
                val currentlyPaused = repository.isPausedFlow.first()
                applyPauseStateChange(targetPaused = !currentlyPaused)
            }
            ServiceCommand.DateOrTimeChanged -> {
                repository.initializeTodayRecord(currentDateProvider())
                updateNotificationForCurrentState()
            }
        }
    }

    private suspend fun processSingleReading(reading: StepSensorReading) {
        hasReceivedReadingSinceListen = true
        val updatedDaily = repository.processHardwareSensorReading(
            rawSensorSteps = reading.totalStepsSinceBoot,
            eventTimestampNanos = reading.timestampNanos
        )
        val isPaused = repository.isPausedFlow.first()
        if (updatedDaily.date == currentDateProvider()) {
            updateNotification(updatedDaily.steps, updatedDaily.goalSteps, isPaused)
        } else {
            updateNotificationForCurrentState()
        }
    }

    private suspend fun drainBufferedSensorReadings() {
        while (true) {
            val buffered = commandChannel.tryReceive().getOrNull() ?: break
            if (buffered is ServiceCommand.SensorReading) {
                processSingleReading(buffered.reading)
            } else {
                deferredCommands.addLast(buffered)
            }
        }
    }

    private suspend fun applyPauseStateChange(targetPaused: Boolean) {
        if (::sensorManager.isInitialized) {
            sensorManager.flushAndWait()
        }
        // Drain any readings delivered by the flush under the previous pause state before flipping isPaused
        drainBufferedSensorReadings()
        val needsBaselineSyncOnNextReading = !targetPaused && !hasReceivedReadingSinceListen
        repository.setPaused(
            isPaused = targetPaused,
            pendingResumeBaseline = needsBaselineSyncOnNextReading
        )
        updateNotificationForCurrentState()
    }

    private fun startTracking(): Boolean {
        if (!isForegroundStarted) {
            val initialNotification = buildNotification(steps = 0, goal = 6000, isPaused = false)
            if (!promoteToForeground(initialNotification)) {
                handleStop()
                return false
            }
            isForegroundStarted = true
        }

        if (!::sensorManager.isInitialized ||
            !sensorManager.isSensorAvailable ||
            !sensorManager.startListening(foregroundMode = isUiBound)
        ) {
            handleStop()
            return false
        }
        hasReceivedReadingSinceListen = false

        commandChannel.trySend(ServiceCommand.Initialize)
        return true
    }

    private fun tryPromoteMinimalForegroundForStop() {
        val notification = buildNotification(steps = 0, goal = 6000, isPaused = false)
        if (promoteToForeground(notification)) {
            isForegroundStarted = true
        }
    }

    private fun handleStop() {
        isForegroundStarted = false
        hasReceivedReadingSinceListen = false
        if (::sensorManager.isInitialized) {
            sensorManager.stopListening()
        }
        ServiceCompat.stopForeground(this, ServiceCompat.STOP_FOREGROUND_REMOVE)
        getSystemService(NotificationManager::class.java)?.cancel(NOTIFICATION_ID)
        stopSelf()
    }

    private fun promoteToForeground(notification: Notification): Boolean {
        val serviceType = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.UPSIDE_DOWN_CAKE) {
            ServiceInfo.FOREGROUND_SERVICE_TYPE_HEALTH
        } else {
            0
        }
        return try {
            ServiceCompat.startForeground(
                this,
                NOTIFICATION_ID,
                notification,
                serviceType
            )
            true
        } catch (_: SecurityException) {
            false
        } catch (_: IllegalStateException) {
            false
        }
    }

    internal suspend fun updateNotificationForCurrentState() {
        if (!isForegroundStarted) return
        val today = currentDateProvider()
        val todayRecord = repository.getDailyRecordDirect(today)
        val profile = repository.userProfileFlow.first()
        val isPaused = repository.isPausedFlow.first()
        val steps = todayRecord?.steps ?: 0
        val goal = profile.dailyGoal
        updateNotification(steps, goal, isPaused)
    }

    internal fun updateNotification(steps: Int, goal: Int, isPaused: Boolean) {
        if (!isForegroundStarted) return
        val notificationManager = getSystemService(NotificationManager::class.java)
        notificationManager?.notify(NOTIFICATION_ID, buildNotification(steps, goal, isPaused))
    }

    internal fun buildNotification(steps: Int, goal: Int, isPaused: Boolean): Notification {
        val launchIntent = Intent(this, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_SINGLE_TOP or Intent.FLAG_ACTIVITY_CLEAR_TOP
        }
        val openAppIntent = PendingIntent.getActivity(
            this,
            0,
            launchIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

            action = if (isPaused) ACTION_RESUME else ACTION_PAUSE
        }
        val actionPendingIntent = PendingIntent.getForegroundService(
            this,
            1,
            actionIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val percent = if (goal > 0) {
            ((steps.toLong() * 100L) / goal.toLong()).coerceIn(0L, 100L).toInt()
        } else {
            0
        }
        val formattedSteps = String.format(Locale.getDefault(), "%,d", steps)
        val formattedGoal = String.format(Locale.getDefault(), "%,d", goal)
        val statusText = if (isPaused) {
            getString(R.string.notification_status_paused, percent)
        } else {
            getString(R.string.notification_status_active, percent, formattedGoal)
        }

            if (isPaused) R.string.notification_action_resume else R.string.notification_action_pause
        )

        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setSmallIcon(R.drawable.ic_tab_today)
            .setContentTitle(getString(R.string.notification_title_steps, formattedSteps))

            .setOngoing(true)
            .setSilent(true)
            .setContentIntent(openAppIntent)
            .addAction(0, actionLabel, actionPendingIntent)
            .build()
    }

package com.nordic.stepcounter.service

import android.content.Context
import android.hardware.Sensor
import android.hardware.SensorEvent
import android.hardware.SensorEventListener2
import android.hardware.SensorManager
import kotlinx.coroutines.CompletableDeferred
import kotlinx.coroutines.sync.Mutex
import kotlinx.coroutines.sync.withLock
import kotlinx.coroutines.withTimeoutOrNull
import kotlin.math.roundToLong

data class StepSensorReading(
    val totalStepsSinceBoot: Long,
    val timestampNanos: Long = 0L
)

/**
 * Low-power hardware step counter manager.
 * Exclusively uses [Sensor.TYPE_STEP_COUNTER] (sensor type 19), switching between
 * immediate delivery while the UI is visible and 60-second hardware FIFO batching in background.
 */
class StepSensorManager(
    private val sensorManager: SensorManager?,
    private val onStepCountChanged: ((StepSensorReading) -> Unit)? = null
) : SensorEventListener2 {

    constructor(
        context: Context,
        onStepCountChanged: ((StepSensorReading) -> Unit)? = null
    ) : this(
        context.getSystemService(Context.SENSOR_SERVICE) as? SensorManager,
        onStepCountChanged
    )

    private val stepCounterSensor: Sensor? =
        sensorManager?.getDefaultSensor(Sensor.TYPE_STEP_COUNTER)

    val isSensorAvailable: Boolean
        get() = stepCounterSensor != null

    @Volatile
    private var isListening = false
    @Volatile
    private var isForegroundMode = false
    private val flushMutex = Mutex()
    @Volatile
    private var flushDeferred: CompletableDeferred<Boolean>? = null

    private fun resolveReportLatencyUs(sensor: Sensor, foregroundMode: Boolean): Int {
        if (foregroundMode) return 0
        val supportsBatching = runCatching { sensor.fifoMaxEventCount > 0 }.getOrDefault(true)
        return if (supportsBatching) BATCH_LATENCY_US else 0
    }

    /**
     * Registers the hardware step counter listener and immediately triggers a FIFO flush
     * so an initial baseline reading is delivered without waiting for a full batch cycle.
     */
    @Synchronized
    fun startListening(foregroundMode: Boolean = isForegroundMode): Boolean {
        val sensor = stepCounterSensor ?: return false
        if (isListening && isForegroundMode == foregroundMode) return true
        if (isListening) {
            sensorManager?.unregisterListener(this)
            isListening = false
        }

        val latencyUs = resolveReportLatencyUs(sensor, foregroundMode)
        val success = sensorManager?.registerListener(
            this,
            sensor,
            SensorManager.SENSOR_DELAY_NORMAL,
            latencyUs
        ) ?: false
        if (success) {
            isListening = true
            isForegroundMode = foregroundMode
            sensorManager.flush(this)
        }
        return success
    }

    /**
     * Updates batching latency between foreground interactive mode (0us) and background batching (60s).
     */
    @Synchronized
    fun setForegroundMode(foregroundMode: Boolean): Boolean {
        if (!isListening) {
            isForegroundMode = foregroundMode
            return false
        }
        if (isForegroundMode == foregroundMode) {
            if (foregroundMode) {
                flush()
            }
            return true
        }
        return startListening(foregroundMode)
    }

    /**
     * Unregisters the sensor listener and completes any pending flush waiter cleanly.
     */
    @Synchronized
    fun stopListening() {
        sensorManager?.unregisterListener(this)
        isListening = false
        flushDeferred?.complete(false)
        flushDeferred = null
    }

    /**
     * Flushes the hardware sensor FIFO buffer to immediately process batched steps.
     * Returns true if flush was initiated, false otherwise.
     */
    fun flush(): Boolean {
        return sensorManager?.flush(this) ?: false
    }

    /**
     * Flushes the hardware sensor FIFO buffer and suspends until [onFlushCompleted] is invoked
     * or [timeoutMs] elapses. Returns true only when flush actually completed before timing out.
     */
    suspend fun flushAndWait(timeoutMs: Long = 500L): Boolean = flushMutex.withLock {
        val deferred = CompletableDeferred<Boolean>()
        flushDeferred = deferred
        try {
            val initiated = sensorManager?.flush(this) ?: false
            if (!initiated) return@withLock false
            withTimeoutOrNull(timeoutMs) {
                deferred.await()
            } ?: false
        } finally {
            if (flushDeferred === deferred) {
                flushDeferred = null
            }
        }
    }

    override fun onFlushCompleted(sensor: Sensor?) {
        flushDeferred?.complete(true)
    }

    override fun onSensorChanged(event: SensorEvent?) {
        if (event == null) return
        if (event.sensor?.type == Sensor.TYPE_STEP_COUNTER) {
            val rawValue = event.values.getOrNull(0) ?: return
            if (!rawValue.isFinite() || rawValue < 0f) return
            val totalStepsSinceBoot = rawValue.roundToLong()
            if (totalStepsSinceBoot < 0L) return
            onStepCountChanged?.invoke(
                StepSensorReading(
                    totalStepsSinceBoot = totalStepsSinceBoot,
                    timestampNanos = event.timestamp
                )
            )
        }
    }

    override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) {
        // Step counter accuracy does not affect cumulative counting
    }

    companion object {
        const val BATCH_LATENCY_US = 60_000_000 // 60 seconds hardware batching latency
    }
}

/**
 * Single-line text that scales down its font size proportionally when needed so values
 * never clip or wrap on compact viewports or high system font-scale settings.
 */
@Composable
fun AutoScaleSingleLineText(
    text: String,
    style: TextStyle,
    color: Color,
    modifier: Modifier = Modifier,
    minFontSizeSp: Float = 11f,
    contentAlignment: Alignment = Alignment.CenterStart
) {
    val textMeasurer = rememberTextMeasurer()
    val density = LocalDensity.current

    BoxWithConstraints(modifier = modifier, contentAlignment = contentAlignment) {
        val maxWidthPx = with(density) { maxWidth.toPx() }
        val baseFontSizeSp = if (style.fontSize.isSp) style.fontSize.value else 16f

        val fittedFontSize = remember(text, style, maxWidthPx, density.fontScale, minFontSizeSp) {
            if (maxWidthPx <= 0f) {
                baseFontSizeSp.sp
            } else {
                val baseLayout = textMeasurer.measure(
                    text = text,
                    style = style,
                    maxLines = 1,
                    softWrap = false
                )
                val measuredWidth = baseLayout.size.width.toFloat()
                if (measuredWidth <= maxWidthPx) {
                    baseFontSizeSp.sp
                } else {
                    val ratio = (maxWidthPx / measuredWidth) * 0.94f
                    val scaledSp = (baseFontSizeSp * ratio).coerceIn(minFontSizeSp, baseFontSizeSp)
                    val secondPass = textMeasurer.measure(
                        text = text,
                        style = style.copy(fontSize = scaledSp.sp),
                        maxLines = 1,
                        softWrap = false
                    )
                    if (secondPass.size.width > maxWidthPx && secondPass.size.width > 0) {
                        val refinedSp = (scaledSp * (maxWidthPx / secondPass.size.width) * 0.96f)
                            .coerceIn(minFontSizeSp, scaledSp)
                        refinedSp.sp
                    } else {
                        scaledSp.sp
                    }
                }
            }
        }

        Text(
            text = text,
            style = style.copy(fontSize = fittedFontSize),
            color = color,
            maxLines = 1,
            softWrap = false,
            overflow = TextOverflow.Ellipsis
        )
    }

/**
 * Computes the true daily average across completed days within the [windowDays] window,
 * counting skipped days (missing rows) as 0 steps while excluding today's still-incomplete row
 * whenever at least one completed prior day exists.
 */
internal fun calculateWindowDailyAverage(
    today: LocalDate,
    windowDays: Int,
    recordsByDate: Map<String, DailyStepEntity>,
    earliestRecordDateStr: String,
    todaySteps: Int
): Int {
    val windowStart = today.minusDays((windowDays - 1).toLong())
    val yesterday = today.minusDays(1)
    val earliest = earliestRecordDateStr
        .takeIf { it.isNotBlank() }
        ?.let { runCatching { LocalDate.parse(it) }.getOrNull() }
        ?: today
    val effectiveStart = if (earliest.isAfter(windowStart)) earliest else windowStart

    if (effectiveStart.isAfter(yesterday)) {
        return todaySteps
    }

    val completedDayCount = (ChronoUnit.DAYS.between(effectiveStart, yesterday) + 1L)
        .coerceAtLeast(1L)
    var totalSteps = 0L
    var cursor = effectiveStart
    while (!cursor.isAfter(yesterday)) {
        totalSteps += (recordsByDate[cursor.toString()]?.steps ?: 0).toLong()
        cursor = cursor.plusDays(1)
    }
    return (totalSteps / completedDayCount).toInt()
}

import android.app.Activity

import androidx.compose.material3.lightColorScheme
import androidx.compose.runtime.Composable
import androidx.compose.runtime.CompositionLocalProvider

import androidx.compose.ui.platform.LocalView
import androidx.core.view.WindowCompat
import com.nordic.stepcounter.data.preferences.AppTheme

@Composable
fun NordicStepTheme(
    appTheme: AppTheme = AppTheme.NORDIC_DARK,
    content: @Composable () -> Unit
) {
    val palette = appTheme.toStepColorPalette()

    val colorScheme = if (appTheme.isDark) {
        darkColorScheme(
            primary = palette.accent,
            onPrimary = palette.background,
            secondary = palette.accent,
            onSecondary = palette.background,
            secondaryContainer = palette.accent.copy(alpha = 0.16f),
            onSecondaryContainer = palette.primaryText,
            background = palette.background,
            onBackground = palette.primaryText,
            surface = palette.surface,
            onSurface = palette.primaryText,
            surfaceVariant = palette.surface,
            onSurfaceVariant = palette.mutedText,
            surfaceContainer = palette.surface,
            outline = palette.border,
            outlineVariant = palette.border
        )
    } else {
        lightColorScheme(
            primary = palette.accent,
            onPrimary = palette.background,
            secondary = palette.accent,
            onSecondary = palette.background,
            secondaryContainer = palette.accent.copy(alpha = 0.16f),
            onSecondaryContainer = palette.primaryText,
            background = palette.background,
            onBackground = palette.primaryText,
            surface = palette.surface,
            onSurface = palette.primaryText,
            surfaceVariant = palette.surface,
            onSurfaceVariant = palette.mutedText,
            surfaceContainer = palette.surface,
            outline = palette.border,
            outlineVariant = palette.border
        )
    }

    val view = LocalView.current
    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as? Activity)?.window ?: return@SideEffect
            val insetsController = WindowCompat.getInsetsController(window, view)
            insetsController.isAppearanceLightStatusBars = !appTheme.isDark
            insetsController.isAppearanceLightNavigationBars = !appTheme.isDark
        }
    }

    CompositionLocalProvider(LocalStepPalette provides palette) {
        MaterialTheme(
            colorScheme = colorScheme,
            typography = SwissTypography,
            content = content
        )
    }
}

    private fun Color.relativeLuminance(): Double {
        fun linearize(channel: Float): Double {
            val c = channel.toDouble()
            return if (c <= 0.03928) c / 12.92 else ((c + 0.055) / 1.055).pow(2.4)
        }
        return 0.2126 * linearize(red) + 0.7152 * linearize(green) + 0.0722 * linearize(blue)
    }

    private fun contrastRatio(fg: Color, bg: Color): Double {
        val l1 = fg.relativeLuminance()
        val l2 = bg.relativeLuminance()
        val lighter = maxOf(l1, l2)
        val darker = minOf(l1, l2)
        return (lighter + 0.05) / (darker + 0.05)
    }
